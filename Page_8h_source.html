56/**
57 * @brief Connects Pages with a NavigationBar: one destination per page,
58 * only the selected page is visible. All pages share the same bounds, so the
59 * screen itself never scrolls; each page scrolls on its own.
60 *
61 * NavigationBar<RGB565> nav(Bounds(0, 264, 240, 56));
62 * PageNavigator<RGB565> pages(app.screen(), nav);
63 * pages.addPage(clockPage);
64 * pages.addPage(settingsPage);
65 * pages.begin(Bounds(0, 40, 240, 224)); // adds nav + pages, builds them
66 * // loop(): if (pages.tick(millis())) app.screen().invalidate();
67 */
68template <typename RGB_T = TINYMD_DEFAULT_RGB_T>
69class PageNavigator {
70 public:
71 PageNavigator(Screen<RGB_T>& screen, NavigationBar<RGB_T>& nav) : screen_(screen), nav_(nav) {}
72
73 /// Called after the page was changed (by the user or show()).
74 std::function<void(int index)> onPageChanged;
75
76 /// Adds a page (before begin()).
77 void addPage(Page<RGB_T>& page) { pages_.push_back(&page); }
78
79 /**
80 * @brief Adds the navigation bar (fixed) and the pages (with the indicated
81 * bounds) to the screen, builds the pages and shows the first one.
82 */
83 void begin(Bounds pageBounds) {
84 for (Page<RGB_T>* page : pages_) nav_.addDestination(page->icon(), page->title());
85 nav_.onChange = [this](int index) { show(index); };
86 screen_.addFixedWidget(nav_);
87 for (Page<RGB_T>* page : pages_) {
88 page->bounds = pageBounds;
89 screen_.addWidget(*page); // gives it the theme
90 page->build();
91 }
92 begun_ = true;
93 current_ = -1;
94 show(0);
95 }
96
97 /// Shows the page with the index.
98 void show(int index) {
99 if (index < 0 || index >= count()) return;
100 if (begun_ && current_ >= 0 && current_ != index) pages_[current_]->onHide();
101 const bool changed = current_ != index;
102 current_ = index;
103 for (size_t i = 0; i < pages_.size(); ++i) pages_[i]->visible = static_cast<int>(i) == index;
104 nav_.setSelectedIndex(index);
105 pages_[current_]->onShow();
106 screen_.invalidate();
107 if (changed && onPageChanged) onPageChanged(index);
108 }
109
110 /// Calls tick() of the visible page: true if it must be redrawn.
111 bool tick(uint32_t nowMs) { return current_ >= 0 && pages_[current_]->tick(nowMs); }
112
113 int current() const { return current_; }
114 int count() const { return static_cast<int>(pages_.size()); }
115 Page<RGB_T>& page(int index) { return *pages_[index]; }
116 Page<RGB_T>& currentPage() { return *pages_[current_]; }
117
118 private:
119 Screen<RGB_T>& screen_;
120 NavigationBar<RGB_T>& nav_;
121 std::vector<Page<RGB_T>*> pages_;

123 bool begun_ = false;
124};
