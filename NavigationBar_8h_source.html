54 tinygpu::IFont<RGB_T>& font = *this->theme().typography.label;
55
56 for (int i = 0; i < count; ++i) {
57 const NavDestination<RGB_T>& dest = destinations_[i];
58 const Bounds slot = slots.itemRect(i, count);
59 const int32_t slotX = slot.x;
60 const int32_t slotWidth = slot.w;
61 const bool selected = i == selectedIndex_;

63 : this->theme().colors.onSurfaceVariant;
64 if (!this->enabled) contentColor = blend(contentColor, this->theme().colors.surface, 0.5f);

66 constexpr int32_t kIconSize = 24;

70 kPillHeight);
71 if (selected) {

73 toPx(pill.h / 2), this->theme().colors.secondaryContainer);

90 bool onGesture(const tinygpu::GestureEvent& event) override {
91 const int count = static_cast<int>(destinations_.size());
92 if (!isTapGesture(event.type) || count == 0) return false;
93 const int32_t slotWidth = this->bounds.w / count;
94 int index = (event.point.x - this->bounds.x) / slotWidth;
95 if (index < 0) index = 0;
96 if (index >= count) index = count - 1;
97 if (index != selectedIndex_) {
98 selectedIndex_ = index;
99 if (onChange) onChange(selectedIndex_);
100 }
101 return true;
102 }
