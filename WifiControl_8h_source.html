19/// Signal strength as 0-4 bars (see drawSignalBars()).
20inline int wifiBars(int rssi) {
21 return rssi >= -55 ? 4 : rssi >= -65 ? 3 : rssi >= -75 ? 2 : rssi >= -85 ? 1 : 0;

24/// Keeps the strongest entry of each SSID (hidden networks are dropped),
25/// sorted by signal strength.
26inline void normalizeNetworks(std::vector<WifiNetwork>& list) {
27 std::vector<WifiNetwork> out;
28 for (const WifiNetwork& n : list) {
29 if (n.ssid.empty()) continue;
30 auto it = std::find_if(out.begin(), out.end(), [&](const WifiNetwork& o) { return o.ssid == n.ssid; });
31 if (it == out.end()) {
32 out.push_back(n);
33 } else if (n.rssi > it->rssi) {
34 *it = n;
35 }
36 }
37 std::sort(out.begin(), out.end(), [](const WifiNetwork& a, const WifiNetwork& b) { return a.rssi > b.rssi; });
38 list = out;
39}
