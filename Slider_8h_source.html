12/**
13 * @brief Draggable track-and-thumb slider over [minValue, maxValue].
14 *
15 * Reports isDraggable() == true so Screen::isDraggableAt() (wired to
16 * GestureDetector::isDraggable) classifies a press-and-move starting on the
17 * slider as GestureType::kDrag rather than kPan/kScroll - and Screen latches
18 * the drag to this widget for its whole lifetime, so the thumb keeps
19 * tracking the finger even past the slider's own bounds. Tapping anywhere
20 * on the track also jumps the thumb there.
21 */

30 }

42 RGB_T thumbColor = this->theme().colors.primary;
43 if (!this->enabled) {

50 const int32_t trackTop = this->bounds.centerY() - trackHeight / 2;
51 const size_t trackRadius = toPx(trackHeight / 2);

87 const float t = (maxValue_ > minValue_) ? (value_ - minValue_) / (maxValue_ - minValue_) : 0.0f;
88 return this->bounds.x + static_cast<int32_t>(t * static_cast<float>(this->bounds.w));
