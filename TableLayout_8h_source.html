6/**
7 * @brief Grid with explicit, independently-sized column widths and row heights.
8 *
9 * Not a widget - see GridLayout.h for the rationale. GridLayout assumes
10 * uniform cells and computes the column count for you; TableLayout is for
11 * dashboards mixing wide/narrow columns or short/tall rows, where you supply
12 * the sizes explicitly:
13 *
14 * int32_t colW[] = {120, 80, 80};
15 * int32_t rowH[] = {48, 96};
