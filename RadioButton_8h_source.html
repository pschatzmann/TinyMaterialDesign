58/// Coordinates a set of RadioButtons so selecting one deselects the rest.
59/// Not itself a Widget - each RadioButton must still be added to the
60/// Screen individually, e.g.:
61///
62/// RadioGroup<RGB565> group;
63/// group.onChange = [](int index) { ... };
64/// for (auto& button : buttons) { screen.addWidget(button); group.addButton(button); }
65template <typename RGB_T = TINYMD_DEFAULT_RGB_T>
66class RadioGroup {
67 public:
68 std::function<void(int)> onChange;
69
70 void addButton(RadioButton<RGB_T>& button) {
71 const int index = static_cast<int>(buttons_.size());
72 buttons_.push_back(&button);
73 button.onSelected = [this, index]() { select(index); };
74 }
75
76 int selectedIndex() const {
77 for (size_t i = 0; i < buttons_.size(); ++i) {
78 if (buttons_[i]->selected()) return static_cast<int>(i);
79 }
80 return -1;
81 }
82
83 private:
84 std::vector<RadioButton<RGB_T>*> buttons_;

87 for (size_t i = 0; i < buttons_.size(); ++i) {
88 buttons_[i]->setSelected(static_cast<int>(i) == index);
89 }
90 if (onChange) onChange(index);
91 }
