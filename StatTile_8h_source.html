9/**
10 * @brief A small tile with one value: a caption at the top, the value in an
11 * accent color and an optional unit / sub text - for dashboards and KPIs.
12 *
13 * StatTile<RGB565> rate(Bounds(8, 40, 72, 46), "RATE", "s/day");
14 * rate.setValue("+4.2");
15 * rate.setAccent(colorFromHex<RGB565>(0x10B981));
16 *
17 * The value is drawn in the title font and falls back to the body and label
18 * fonts if it does not fit. Outlined (default, on the page) or tonal (filled
19 * with surfaceVariant, e.g. inside a Card).
20 */

43 void draw(tinygpu::ISurface<RGB_T>& target) override {
44 const MaterialTheme<RGB_T>& t = this->theme();
45 const Bounds& b = this->bounds;
46 const RGB_T bg = tonal ? t.colors.surfaceVariant : t.colors.surface;
47 const size_t radius = toPx(t.shape.medium);
48 target.fillRoundRect(toPx(b.x), toPx(b.y), toPx(b.w), toPx(b.h), radius, bg);
49 if (!tonal) target.drawRoundRect(toPx(b.x), toPx(b.y), toPx(b.w), toPx(b.h), radius, t.colors.outline);
50
51 tinygpu::IFont<RGB_T>& small = *t.typography.label;
52 const int32_t x = b.x + 5;
53 small.drawText(target, static_cast<int16_t>(x), static_cast<int16_t>(b.y + 4), caption_.c_str(),
54 t.colors.onSurfaceVariant, bg, false);
55
56 // the largest font that fits
57 tinygpu::IFont<RGB_T>* fonts[3] = {fontForTypographyRole(valueRole, t), t.typography.body, t.typography.label};
58 tinygpu::IFont<RGB_T>* font = fonts[2];
59 for (tinygpu::IFont<RGB_T>* f : fonts) {
60 if (static_cast<int32_t>(f->measureTextWidth(value_.c_str())) <= b.w - 8) {
61 font = f;
62 break;
63 }
64 }
65 const int32_t smallH = static_cast<int32_t>(small.getHeight(1));
66 const int32_t valueY = sub_.empty() ? b.bottom() - 5 - static_cast<int32_t>(font->getHeight(1))
67 : b.y + 4 + smallH + 3;
68 font->drawText(target, static_cast<int16_t>(x), static_cast<int16_t>(valueY), value_.c_str(),
69 hasAccent_ ? accent_ : t.colors.primary, bg, false);
70 if (!sub_.empty()) {
71 small.drawText(target, static_cast<int16_t>(x), static_cast<int16_t>(b.bottom() - 4 - smallH), sub_.c_str(),
72 t.colors.onSurfaceVariant, bg, false);
73 }
74 }
