12/**
13 * @brief Tappable card with a thumbnail image and a caption - e.g. one tile
14 * in a grid of stations/genres/countries. Pair with GridLayout
15 * (Core/GridLayout.h) to lay out several of these.
16 *
17 * The image is any TinyGPU ISurface<RGB_T> (a Sprite, a decoded BMP, ...)
18 * you already have - this widget only blits it via ISurface::drawSprite(),
19 * it doesn't fetch or decode anything itself (TinyGPU has no HTTP/network
20 * layer). Not owned: pass a pointer that outlives the card, or nullptr for
21 * a placeholder fill. There's no scaling either - drawSprite() blits 1:1,
22 * so pre-size the image to fit the card's image area (bounds minus the
23 * caption strip and padding) before handing it over.
24 */

45 void draw(tinygpu::ISurface<RGB_T>& target) override {
46 const size_t radius = toPx(this->theme().shape.medium);
47 RGB_T background = selected_ ? this->theme().colors.secondaryContainer : this->theme().colors.surface;
48 RGB_T outline = selected_ ? this->theme().colors.primary : this->theme().colors.outline;
49 RGB_T captionColor = selected_ ? this->theme().colors.onSecondaryContainer : this->theme().colors.onSurface;
50 if (!this->enabled) {
51 background = blend(background, this->theme().colors.surface, 0.5f);
52 outline = blend(outline, this->theme().colors.surface, 0.5f);
53 captionColor = blend(captionColor, this->theme().colors.surface, 0.5f);
54 }
55
56 target.fillRoundRect(toPx(this->bounds.x), toPx(this->bounds.y), toPx(this->bounds.w),
57 toPx(this->bounds.h), radius, background);
58 target.drawRoundRect(toPx(this->bounds.x), toPx(this->bounds.y), toPx(this->bounds.w),
59 toPx(this->bounds.h), radius, outline);
60
61 const int32_t pad = 4;
62 const int32_t captionHeight = 18;
63 const Bounds imageArea(this->bounds.x + pad, this->bounds.y + pad, this->bounds.w - 2 * pad,
64 this->bounds.h - captionHeight - 2 * pad);
65
66 if (image_ != nullptr) {
67 target.drawSprite(toPx(imageArea.x), toPx(imageArea.y), *image_);
68 } else if (imageArea.w > 0 && imageArea.h > 0) {
69 target.fillRect(toPx(imageArea.x), toPx(imageArea.y), toPx(imageArea.w), toPx(imageArea.h),
70 this->theme().colors.surfaceVariant);
71 }
72
73 tinygpu::IFont<RGB_T>& font = *this->theme().typography.label;
74 const size_t textWidth = font.measureTextWidth(caption_.c_str());
75 const int32_t textX = this->bounds.centerX() - static_cast<int32_t>(textWidth) / 2;
76 const int32_t textY = this->bounds.bottom() - captionHeight - pad + 2;
77 font.drawText(target, static_cast<int16_t>(textX), static_cast<int16_t>(textY),
78 caption_.c_str(), captionColor, background, false);
79 }
