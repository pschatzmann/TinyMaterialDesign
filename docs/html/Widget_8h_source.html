12/// True for GestureType::kTap and also kDoubleTap - widgets that react to a
13/// plain tap should check this instead of comparing against kTap directly.
14///
15/// Why: GestureDetector reclassifies a tap as kDoubleTap whenever it lands
16/// within doubleTapMaxGapMs (350ms by default) of *any* previous tap, not
17/// just a second tap on the same widget - so quickly tapping one key then
18/// another on Keyboard (completely normal while typing) can turn the second
19/// one into a kDoubleTap. A widget that only matches kTap exactly then
20/// silently drops that keypress, which is exactly the "the Done key doesn't
21/// work reliably" symptom this fixes. None of these widgets give a tap and
22/// a double-tap different meaning, so treating both as "the user tapped me"
23/// is the correct behavior everywhere a plain tap is handled.
24inline bool isTapGesture(tinygpu::GestureType type) {
25 return type == tinygpu::GestureType::kTap || type == tinygpu::GestureType::kDoubleTap;
26}

28/// Darkens every pixel already drawn across `target`'s full extent, in
29/// place - the modal scrim look shared by Dialog and Drawer. Reads each
30/// pixel back and blends it 35% toward black, rather than painting a
31/// flat precomputed color over everything: this format has no alpha
32/// channel, so a flat fill would just erase whatever's actually behind
33/// the modal (the real screen content) instead of dimming it. Relies on
34/// `target` already holding that real content when called - true for
35/// Screen::draw()'s single-buffer pass (widgets are drawn before the
36/// modal), and for Screen::drawDirect()'s modal band loop once it draws
37/// the regular content into each band first (see drawModalDirect()).
38template <typename RGB_T>
39void drawScrim(tinygpu::ISurface<RGB_T>& target) {
40 const size_t w = target.width();
41 const size_t h = target.height();
42 for (size_t y = 0; y < h; ++y) {
43 for (size_t x = 0; x < w; ++x) {
44 target.setPixel(x, y, blend(target.getPixel(x, y), RGB_T(0, 0, 0), 0.35f));
45 }
46 }
47}
