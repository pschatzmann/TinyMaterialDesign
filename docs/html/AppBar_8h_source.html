12 * @brief Top app bar: a title plus optional leading/trailing icon widgets
13 * (typically IconButton).
14 *
15 * `leading`/`trailing` are children AppBar forwards draw()/update()/

53 void draw(tinygpu::ISurface<RGB_T>& target) override {
54 const RGB_T background = hasColorOverride_ ? backgroundOverride_ : this->theme().colors.surface;
55 const RGB_T foreground = hasColorOverride_ ? foregroundOverride_ : this->theme().colors.onSurface;
56
57 target.fillRect(toPx(this->bounds.x), toPx(this->bounds.y), toPx(this->bounds.w),
58 toPx(this->bounds.h), background);
59
60 tinygpu::IFont<RGB_T>& font = *this->theme().typography.title;
61 const size_t textHeight = font.getHeight(1);
62 const int32_t textY = this->bounds.centerY() - static_cast<int32_t>(textHeight) / 2;
63 const int32_t textX = leading != nullptr ? leadingRect().right() + 8 : this->bounds.x + 16;
64 font.drawText(target, static_cast<int16_t>(textX), static_cast<int16_t>(textY),
65 title_.c_str(), foreground, background, false);
66
67 // See Widget::setThemeTint() - lets e.g. a standard-variant IconButton
68 // (whose own default color is tuned for theme.colors.surface, not
69 // whatever this bar's background actually is) read clearly against
70 // this bar by default, without needing its own explicit
71 // setColorOverride() call in the sketch. A no-op for any child that
72 // doesn't override setThemeTint(), and loses to that child's own
73 // explicit override when it has one (e.g. a status-dependent icon).
74 //
75 // leading/trailing are plain public pointer fields the sketch assigns
76 // directly (no registration hook to catch that assignment - see the
77 // class comment), so the theme is pushed into them lazily here, right
78 // before each draw(), rather than once at assignment time.
79 if (leading != nullptr && leading->visible) {
80 leading->setTheme(this->theme());
81 leading->setThemeTint(background, foreground);
82 leading->draw(target);
83 }
84 if (trailing != nullptr && trailing->visible) {
85 trailing->setTheme(this->theme());
86 trailing->setThemeTint(background, foreground);
87 trailing->draw(target);
88 }
89 }

91 bool onGesture(const tinygpu::GestureEvent& event) override {
