7/// Draws a line `thickness` times, offset by whole pixels, as a cheap
8/// approximation of a stroked line - good enough at the small sizes these
9/// vector icons are drawn at (checkbox/radio/icon-button glyphs).
10template <typename RGB_T>
11void drawLineThick(tinygpu::ISurface<RGB_T>& target, int32_t x0, int32_t y0,
12 int32_t x1, int32_t y1, RGB_T color, uint8_t thickness = 2) {
13 if (thickness == 0) thickness = 1;
14 const int32_t half = thickness / 2;
15 for (int32_t o = -half; o < static_cast<int32_t>(thickness) - half; ++o) {
16 target.drawLine(toPx(x0 + o), toPx(y0), toPx(x1 + o), toPx(y1), color);
17 target.drawLine(toPx(x0), toPx(y0 + o), toPx(x1), toPx(y1 + o), color);
18 }
19}
