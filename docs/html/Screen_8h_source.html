14/**
15 * @brief Owns a set of widgets, draws them, and routes gesture events to
16 * whichever one should handle each event - the one root Container every
17 * sketch has, plus what only a root needs: a second non-scrolling widget
18 * layer, modal dialog presentation, and actually writing to a display.
19 *
20 * Screen does not own the widgets (the sketch does, typically as globals -
21 * the same non-owning-reference composition TinyGPU itself uses everywhere,
22 * e.g. DeviceOutput(driver&)). Usage, mirroring TinyGPU's own examples:
23 *
24 * Screen<RGB565> screen(theme);
25 * Button<RGB565> okButton(...);
26 * screen.addWidget(okButton);
27 *
28 * GestureDetector gestures;
29 * gestures.onGesture = [](GestureEvent& e) { screen.handleGesture(e); };
30 * gestures.isDraggable = [](int16_t x, int16_t y) {
31 * return screen.isDraggableAt(x, y);
32 * };
33 *
34 * void loop() {
35 * gestures.update(touchDriver);
36 * screen.update(millis());
37 * if (screen.isDirty()) { // see isDirty() - skips the (slow)
38 * screen.draw(surface); // redraw + display write when
39 * display.writeData(surface); // nothing changed this frame
40 * }
41 * }
42 *
43 * Vertical scrolling: widgets added via addWidget() are Container's own
44 * children (see Container.h) - laid out in one "content" coordinate space
45 * that can be taller than the surface passed to draw() - if it is, dragging
46 * up/down over any non-interactive spot (an empty gap, a Label, a Card, ...)
47 * scrolls that content, the same gesture TouchDriver/GestureDetector already
48 * classify as kScroll/kPan. Widgets that should stay put while the content
49 * behind them scrolls - a top AppBar, a bottom Keyboard - go through
50 * addFixedWidget() instead; they're always drawn and hit-tested on top of
51 * the scrollable content, at their own authored (screen-space) bounds. If
52 * everything already fits within the drawn surface, scrolling never
53 * engages (offset stays 0), no scrollbar is drawn, and existing sketches
54 * that only call addWidget() are unaffected. A thin scrollbar on the right
55 * edge appears automatically whenever there's something to scroll - see
56 * Container::draw().
57 *
58 * Since Screen *is* a Container (its scrollable content is just
59 * Container's own child list), an addWidget()'d widget can itself be a
60 * Container - nesting a smaller scrollable panel inside the one root
61 * scrollable area. See Container.h for that case's own scroll/gesture
62 * behavior and limitations (which Screen shares for its own content, listed
63 * below).
64 *
65 * A widget that straddles the viewport's top/bottom edge while scrolled
66 * partway off it is cropped to the viewport (see ISurface::pushClipRect(),
67 * used the same way Container.h uses it for its own children) rather than
68 * rendered pinned to y=0 at full size or skipped outright.
69 */

180 /// Renders the full layout straight to `driver` without ever needing a
181 /// full-screen framebuffer - for boards that can't spare a single
182 /// contiguous allocation the size of the whole screen (e.g. classic
183 /// ESP32 without PSRAM: a 240x320 RGB565 buffer is 153,600 bytes, which
184 /// can exceed the largest free block even with plenty of *total* free
185 /// heap, since the heap ends up split into multiple smaller regions).
186 ///
187 /// Each visible widget - scrollable or fixed, skipping anything
188 /// invisible or entirely outside the viewport (including scrolled off
189 /// it), same idea as content that's never drawn today - gets `scratch`
190 /// resized to exactly its own bounds, is drawn into it once, and is
191 /// blitted directly to the panel at its own position: genuinely one
192 /// small buffer and one write per control, not a shared full-screen (or
193 /// full-band) one. `scratch` is one small, reused, real
194 /// (non-owning, already-begin()'d) surface - since Surface::resize()
195 /// only reallocates when growing past its current capacity, repeatedly
196 /// resizing it this way settles down to zero reallocation once it's
197 /// been sized to the largest single widget this layout ever draws.
198 ///
199 /// The one case that can't work this way is a presented modal
200 /// (Dialog/Drawer): its scrim always covers the *full* viewport
201 /// regardless of the modal's own bounds (see Dialog.h/Drawer.h), and it
202 /// can itself span most of the screen (a full-height Drawer) - both too
203 /// much for one small per-widget buffer. So while a modal is presented,
204 /// everything else is skipped outright (it would just be fully
205 /// occluded) and only the modal is drawn: its drawBackground() (see
206 /// Widget.h) - scrim plus whatever chrome isn't one of its children -
207 /// one short horizontal strip at a time via a WindowedSurface (see that
208 /// class) that reports the full viewport size for the scrim's own math
209 /// but clips actual pixel writes to the current strip, followed by each
210 /// of its children (see childCount()/child()) individually via the same
211 /// per-widget path as everything else - e.g. Drawer's own list items,
212 /// each drawn once rather than redrawn on every strip. A modal type
213 /// that doesn't override drawBackground()/childCount() (Dialog, as yet)
214 /// falls back to the default drawBackground() = draw() and childCount()
215 /// = 0, i.e. its *entire* contents (chrome + actions) drawn banded, same
216 /// as before.
217 ///
218 /// This is an alternative to draw()+writeData(surface) (unaffected by
219 /// this method's existence) - pick whichever fits your board.
220 void drawDirect(tinygpu::DisplayDriver<RGB_T>& driver, tinygpu::ISurface<RGB_T>& scratch,
221 int32_t viewportWidth, int32_t viewportHeight) {
222 bounds.w = viewportWidth;
223 bounds.h = viewportHeight;
224 this->clampScroll();
225
226 if (dialog_ != nullptr && dialog_->visible) {
227 drawModalDirect(driver, scratch, viewportWidth, viewportHeight);
228 dirty_ = false;
229 return;
230 }

232 const RGB_T background = hasBackgroundColor_ ? backgroundColor_ : ownedTheme_.colors.background;
233
234 // Unlike draw() (which clears the whole target surface up front - see
235 // its target.clear() call above), per-widget rendering only ever
236 // touches each widget's own small bounds - nothing paints the gaps
237 // between/around them otherwise, which would just show whatever was
238 // on the panel before (stale content, or black on first boot).
239 // writeColor() is DisplayDriver's own cheap one-row-buffer solid
240 // fill (see its doc comment) - no full-viewport buffer needed here
241 // either.
242 driver.writeColor(static_cast<size_t>(viewportWidth), static_cast<size_t>(viewportHeight),
243 background);
244
245 const int scrollCount = this->effectiveCount();
246 for (int i = 0; i < scrollCount; ++i) {
247 Widget<RGB_T>* widget = this->effectiveChild(i);
248 if (!widget->visible) continue;
249 Bounds screenBounds = widget->bounds;
250 screenBounds.y -= this->scrollOffset_;
251 if (!intersectsViewport(screenBounds, viewportWidth, viewportHeight)) continue;
252 widget->bounds.y -= this->scrollOffset_;
253 drawWidgetDirect(driver, scratch, *widget, background);
254 widget->bounds.y += this->scrollOffset_;
255 }
256 drawScrollbarDirect(driver, scratch, viewportWidth, viewportHeight);
257 for (Widget<RGB_T>* widget : fixedWidgets_) {
258 if (!widget->visible) continue;
259 if (!intersectsViewport(widget->bounds, viewportWidth, viewportHeight)) continue;
260 drawWidgetDirect(driver, scratch, *widget, background);
261 }
262 dirty_ = false;
263 }

276 /// draggable widget (currently only Slider) is classified as kDrag

290 bool onGesture(const tinygpu::GestureEvent& event) override {
291 using tinygpu::GesturePhase;
292 using tinygpu::GestureType;
293
294 // Any gesture reaching here might change something on screen (a ripple
295 // starting, a drag position, a scroll offset, or anything a click/
296 // change callback touches) - see isDirty(). Coarse but cheap, and only
297 // as frequent as actual input, not the loop() rate.
298 dirty_ = true;
299
300 if (dialog_ != nullptr) {
301 // Modal: every gesture goes to the presented dialog while it's shown.
302 // Dialog bounds are always screen-space, so no offset applies.
303 const bool handled = dialog_->enabled ? dialog_->onGesture(event) : false;
304 if (Container<RGB_T>::isContinuousType(event.type) && event.phase == GesturePhase::kEnded) {
305 activeFixedWidget_ = nullptr;
306 usingFixedWidget_ = false;
307 this->activeChild_ = nullptr;
308 this->scrollDragActive_ = false;
309 }
310 return handled;
311 }
312
313 if (Container<RGB_T>::isContinuousType(event.type)) {
314 // Drag-like gestures: latch what kBegan hit and keep routing to it
315 // for kChanged/kEnded even if the pointer leaves its bounds -
316 // otherwise dragging a Slider thumb past its own bounds would stop
317 // updating it. kPan/kScroll only ever fire over a non-draggable
318 // start point (see GestureDetector::classifyDragType), so they
319 // always mean "scroll the content" and are always left to
320 // Container::onGesture() below rather than the fixed layer. Fixed
321 // widgets - always screen-space, never scrollable - take priority
322 // over Container's own (scrollable) children, same order as
323 // hitTestFixed()-then-Container in the discrete branch below.
324 if (event.phase == GesturePhase::kBegan) {
325 usingFixedWidget_ = false;
326 activeFixedWidget_ = nullptr;
327 if (event.type != GestureType::kPan && event.type != GestureType::kScroll) {
328 Widget<RGB_T>* hit = hitTestFixed(event.startPoint.x, event.startPoint.y);
329 if (hit != nullptr) {
330 usingFixedWidget_ = true;
331 activeFixedWidget_ = hit;
332 }
333 }
334 }
335
336 bool handled = false;
337 if (usingFixedWidget_) {
338 if (activeFixedWidget_ != nullptr && activeFixedWidget_->enabled) {
339 handled = activeFixedWidget_->onGesture(event);
340 }
341 } else {
342 handled = Container<RGB_T>::onGesture(event);
343 }
344
345 if (event.phase == GesturePhase::kEnded) {
346 activeFixedWidget_ = nullptr;
347 usingFixedWidget_ = false;
348 }
349 return handled;
350 }
351
352 // Discrete gestures (tap/double-tap/long-press/swipe-*) always report
353 // phase kEnded with no preceding kBegan, so they're hit-tested fresh
354 // at the event's own point. Fixed widgets are checked first since
355 // they're always drawn over scrollable ones (matches draw()); falling
356 // through to Container::onGesture() covers this screen's own scrollable
357 // content, including translating the point into content space.
358 Widget<RGB_T>* hit = hitTestFixed(event.point.x, event.point.y);
359 if (hit != nullptr && hit->enabled) return hit->onGesture(event);
360 return Container<RGB_T>::onGesture(event);
361 }

380 static bool intersectsViewport(const Bounds& bounds, int32_t viewportWidth, int32_t viewportHeight) {

385 /// Last-added widget wins ties, i.e. later addFixedWidget() calls are
386 /// treated as drawn on top (matches draw()). (x, y) is screen-space,
387 /// same as every fixed widget's own bounds.
388 Widget<RGB_T>* hitTestFixed(int32_t x, int32_t y) const {
389 for (auto it = fixedWidgets_.rbegin(); it != fixedWidgets_.rend(); ++it) {
390 Widget<RGB_T>* widget = *it;
391 if (widget->visible && widget->enabled && widget->bounds.contains(x, y)) return widget;
392 }

396 /// Draws one widget into `scratch` (resized to exactly its own bounds)
397 /// and blits it to `driver` at its own position - see drawDirect().
398 /// No-ops (leaving the display unwritten) if `widget`'s bounds are
399 /// empty or scratch.resize() fails.
400 void drawWidgetDirect(tinygpu::DisplayDriver<RGB_T>& driver, tinygpu::ISurface<RGB_T>& scratch,
401 Widget<RGB_T>& widget, RGB_T background) {
402 const int32_t x = widget.bounds.x;
403 const int32_t y = widget.bounds.y;
404 const int32_t w = widget.bounds.w;
405 const int32_t h = widget.bounds.h;
406 if (w <= 0 || h <= 0) return;
407 if (!scratch.resize(static_cast<size_t>(w), static_cast<size_t>(h))) return;
408
409 tinygpu::WindowedSurface<RGB_T> window(scratch, x, y, static_cast<size_t>(w),
410 static_cast<size_t>(h), scratch.font());
411 window.clear(background);
412 widget.draw(window);
413 driver.writeData(scratch, x, y);
414 }

416 /// Draws the right-edge scroll position indicator directly - mirrors
417 /// Container::drawScrollbar(), just resized/blitted instead of drawn
418 /// into the shared full-size target. Only kBarWidth px wide regardless
419 /// of viewport height, so - unlike a modal - this always fits in one
420 /// shot.
421 void drawScrollbarDirect(tinygpu::DisplayDriver<RGB_T>& driver, tinygpu::ISurface<RGB_T>& scratch,
422 int32_t viewportWidth, int32_t viewportHeight) {
423 const int32_t content = this->contentHeight();
424 if (content <= viewportHeight) return;
425
426 constexpr int32_t kBarWidth = 4;
427 const int32_t trackX = viewportWidth - kBarWidth;
428 if (!scratch.resize(static_cast<size_t>(kBarWidth), static_cast<size_t>(viewportHeight))) return;
429
430 tinygpu::WindowedSurface<RGB_T> window(scratch, trackX, 0, static_cast<size_t>(kBarWidth),
431 static_cast<size_t>(viewportHeight), scratch.font());
432 window.fillRect(trackX, 0, kBarWidth, viewportHeight,
433 blend(ownedTheme_.colors.surface, ownedTheme_.colors.outline, 0.25f));
434
435 const int32_t maxScroll = content - viewportHeight;
436 const int32_t thumbHeight =
437 std::max(int32_t(24), viewportHeight * viewportHeight / content);
438 const int32_t thumbY = maxScroll > 0
439 ? (this->scrollOffset_ * (viewportHeight - thumbHeight)) / maxScroll
440 : 0;
441 window.fillRoundRect(trackX, thumbY, kBarWidth, thumbHeight, kBarWidth / 2, ownedTheme_.colors.primary);
442 driver.writeData(scratch, trackX, 0);
443 }

445 /// Draws the presented modal - its drawBackground() banded (see
446 /// drawDirect() for why), then each of its children (if any)
447 /// individually via drawWidgetDirect(). `scratch` ends this method
448 /// resized to `viewportWidth x kModalBandHeight`.
449 void drawModalDirect(tinygpu::DisplayDriver<RGB_T>& driver, tinygpu::ISurface<RGB_T>& scratch,
450 int32_t viewportWidth, int32_t viewportHeight) {
451 const RGB_T background = hasBackgroundColor_ ? backgroundColor_ : ownedTheme_.colors.background;
452
453 constexpr int32_t kModalBandHeight = 40;
454 if (scratch.resize(static_cast<size_t>(viewportWidth), static_cast<size_t>(kModalBandHeight))) {
455 for (int32_t bandY = 0; bandY < viewportHeight; bandY += kModalBandHeight) {
456 tinygpu::WindowedSurface<RGB_T> window(scratch, 0, bandY,
457 static_cast<size_t>(viewportWidth),
458 static_cast<size_t>(viewportHeight),
459 scratch.font());
460 // drawBackground() darkens whatever's already in `window` (see
461 // Widget::drawScrim()) rather than painting a flat scrim color -
462 // deliberately just a clean background here, not the real
463 // regular content, on this memory/CPU-constrained rendering
464 // path (see drawDirect()'s doc comment) - without this clear()
465 // it would darken whatever stale pixels `scratch` happened to
466 // still hold from the last (differently-sized) widget drawn
467 // into it.
468 window.clear(background);
469 dialog_->drawBackground(window);
470 driver.writeData(scratch, 0, bandY);
471 }
472 }
473
474 const int childCount = dialog_->childCount();
475 for (int i = 0; i < childCount; ++i) {
476 Widget<RGB_T>* child = dialog_->child(i);
477 if (child != nullptr && child->visible) {
478 drawWidgetDirect(driver, scratch, *child, background);
479 }
480 }
481 }
