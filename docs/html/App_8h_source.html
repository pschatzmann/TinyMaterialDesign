12/**
13 * @brief Bundles the surface + display + gesture routing + Screen
14 * boilerplate that every single-widget example under examples/controls and
15 * examples/layouts otherwise hand-rolls identically, down to a constructor
16 * plus two calls: begin() and update().
17 *
18 * Without `App`, every one of those sketches repeats the same ~15 lines:
19 *
20 * Surface<RGB565> surface(kWidth, kHeight, FontRGB565);
21 * DeviceOutput<RGB565> display(board.display());
22 * GestureDetector gestures;
23 * Screen<RGB565> screen(theme);
24 * // ...then in setup(): display.begin(); surface.begin();
25 * // gestures.onGesture = ...; gestures.isDraggable = ...;
26 * // ...then in loop(): gestures.update(*board.touch()); screen.update(...);
27 * // if (screen.isDirty()) { screen.draw(surface); display.writeData(surface); }
28 *
29 * `App` is exactly that, reduced to:
30 *
31 * LCDBoardGuitionESP32_LVGL_2_4Display board;
32 * App<RGB565> app(board);
33 * Button<RGB565> myButton(Bounds(...), "Tap me");
34 *
35 * void setup() {
36 * Serial.begin(115200);
37 * app.begin();
38 * myButton.onClick = []() { ... };
39 * app.screen().addWidget(myButton);
40 * }
41 *
42 * void loop() {
43 * app.update();
44 * }
45 *
46 * The board itself is left to the sketch, not `App` - board *types* differ
47 * per target (a desktop board takes width/height in its constructor, most
48 * real hardware boards take none, and a board with nonstandard wiring
49 * takes pin numbers) enough that there's no one constructor `App` could
50 * call on your behalf - `TinyMaterialDesignConfig.h`'s `DefaultBoard` is a
51 * ready-made object of whichever concrete type fits the current platform,
52 * for exactly this reason (see its own doc comment); pass a different
53 * `LCDBoard` instead for anything else. Whatever you pass must already be
54 * fully constructed and must outlive this `App` - the same
55 * non-owning-reference convention every other TinyMaterialDesign class
56 * uses. Width/height are read from `board.width()`/`board.height()`, so
57 * they don't need to be passed again.
58 *
59 * Pinned to `LCDBoard`'s own RGB565-only `display()` (see
60 * `TinyGPU/Boards/LCDBoardsCommon.h`'s class comment) - `RGB_T` is
61 * templated for consistency with every other class in this library, but in
62 * practice only `RGB565` compiles today, the same constraint every
63 * hand-rolled example already has.
64 *
65 * begin() falls back to Screen::drawDirect()'s small-per-widget-buffer
66 * rendering automatically if the full-screen framebuffer `surface()` needs
67 * can't be allocated (see that method's own doc comment for why this can
68 * happen even with plenty of free heap) - a board without enough RAM for a
69 * full framebuffer draws exactly the same widgets through the same
70 * `screen()` API, just via a smaller reused buffer. See usesDirectRender().
71 *
72 * Only one `App` may be alive at a time (per `RGB_T` instantiation): the
73 * gesture callbacks it wires up in begin() are plain C function pointers
74 * (`GestureDetector::onGesture`/`isDraggable`, not `std::function`), so
75 * they can't capture `this` - `App` routes them through a static
76 * self-pointer instead. Every example this replaces already only ever
77 * created one `Screen`/`GestureDetector` pair as globals, so this isn't a
78 * new restriction in practice.
79 */

81class App {

83 /// `board` must already be fully constructed (with whatever constructor
84 /// arguments its concrete type needs) and must outlive this `App` - see
85 /// the class comment.
86 explicit App(tinygpu::LCDBoard& board, const MaterialTheme<RGB_T>& theme = defaultTheme<RGB_T>())
87 : board_(board),
88 surface_(board.width(), board.height(), tinygpu::FontRGB565),
89 scratch_(1, 1, tinygpu::FontRGB565),
90 display_(board.display()),
91 screen_(theme) {
92 instance_ = this;
93 }
94
95 /// Starts the board, display, and surface, and wires gesture routing to
96 /// the Screen - call once from setup() (after Serial.begin(115200) -
97 /// see the class comment), before or after registering widgets via
98 /// screen() (order between the two doesn't matter).
99 ///
100 /// If the full-screen framebuffer `surface()` needs can't be allocated
101 /// (see Screen::drawDirect()'s doc comment - typically a classic ESP32
102 /// without PSRAM, where a contiguous block that size may not exist even
103 /// with plenty of free heap overall), `update()` transparently falls
104 /// back to drawDirect() instead: same widgets, same screen() API, just a
105 /// small reused per-widget buffer rather than one full-screen one.
106 void begin() {
107 board_.begin();
108 display_.begin();
109 directRender_ = !surface_.begin();
110 if (directRender_) {
111 Serial.println(
112 "App: full-screen framebuffer allocation failed - falling back to "
113 "Screen::drawDirect()");
114 scratch_.begin();
115 }
116 gestures_.onGesture = &App::onGestureTrampoline;
117 gestures_.isDraggable = &App::isDraggableTrampoline;
118 }
119
120 /// Call once per loop(): polls touch input, advances widget animations,
121 /// and redraws only if something actually changed - see
122 /// Screen::isDirty().
123 void update() {
124 gestures_.update(*board_.touch());
125 screen_.update(millis());
126 if (screen_.isDirty()) {
127 if (directRender_) {
128 screen_.drawDirect(board_.display(), scratch_, static_cast<int32_t>(width()),
129 static_cast<int32_t>(height()));
130 } else {
131 screen_.draw(surface_);
132 display_.writeData(surface_);
133 }
134 }
135 }
136
137 /// True once begin() has fallen back to the low-RAM Screen::drawDirect()
138 /// path because surface()'s full-screen framebuffer couldn't be
139 /// allocated - see begin()'s doc comment. Mainly useful for logging/
140 /// diagnostics; sketches don't need to branch on it themselves.
141 bool usesDirectRender() const { return directRender_; }
142
143 /// Register widgets here - addWidget()/addFixedWidget()/presentDialog(),
144 /// same as any other sketch's `screen`.
145 Screen<RGB_T>& screen() { return screen_; }
146
147 /// The theme in effect - see Screen::theme(). Handy for reading a color
148 /// role directly (e.g. for a widget's own setColorOverride()) without
149 /// keeping a separate MaterialTheme alongside `App`.
150 const MaterialTheme<RGB_T>& theme() const { return screen_.theme(); }
151
152 /// Rarely needed directly (App's own update() already draws to it) -
153 /// exposed for the odd sketch that wants to draw something outside the
154 /// widget tree. Empty/unused while usesDirectRender() is true - update()
155 /// draws to the display through `scratch` instead in that case.
156 tinygpu::Surface<RGB_T>& surface() { return surface_; }
157
158 /// The display's width/height in pixels - straight from the board (see
159 /// its own constructor), so a sketch never needs to declare its own
160 /// `kWidth`/`kHeight` just to lay out widgets; use `app.width()`/
161 /// `app.height()` directly wherever a widget's `Bounds` needs them.
162 size_t width() const { return board_.width(); }
163 size_t height() const { return board_.height(); }
164
165 private:
166 static void onGestureTrampoline(tinygpu::GestureEvent& event) { instance_->screen_.handleGesture(event); }
167 static bool isDraggableTrampoline(int16_t x, int16_t y) {
168 return instance_->screen_.isDraggableAt(x, y);
169 }
