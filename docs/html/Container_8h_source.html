12/**
13 * @brief A widget that holds child widgets, draws/dispatches gestures to
14 * them, and scrolls its content vertically when it overflows its own bounds.
15 *
16 * This is the nestable counterpart to Screen (which does the same job once,
17 * for the one root every sketch already has) - unlike GridLayout/LinearLayout/
18 * etc. (pure rect calculators consumed before construction), Container is a
19 * real Widget: it owns a child list, and since it's a Widget itself it can be
20 * added as a child of another Container, nesting arbitrarily deep.
21 *
22 * Container<RGB565> panel(Bounds(16, 56, 288, 300));
23 * panel.addChild(card1);
24 * panel.addChild(card2); // more content than panel.bounds.h -> scrolls
25 * screen.addWidget(panel);
26 *
27 * Children are non-owning references (same convention as every other
28 * composite in this library - Dialog's actions, Drawer's items, ...) and are
29 * authored with absolute Bounds starting at/near this container's own
30 * bounds.y, exactly like Screen's own scrollable content: at scrollOffset 0
31 * a child renders at its own authored position, and scrolling shifts that
32 * the same way Screen::draw() does.
33 *
34 * Children partially or fully scrolled past this container's own bounds
35 * are cropped to it (see ISurface::pushClipRect()), not just skipped or
36 * drawn in full - matching Screen's own root-level clip.
37 *
38 * Known limitation (shared with Screen - see its class comment): scrolling
39 * is vertical only, one axis.
40 *
41 * Alternative to addChild(): setChildProvider(count, at) switches this
42 * container to callback-driven content, for lists too large to keep every
43 * item's Widget resident at once (thousands of rows) - `count` reports how
44 * many logical items there are, `at(index)` returns a *reference* to the
45 * Widget representing item `index`, typically a small pool of real Widgets
46 * reused/repositioned/relabeled for whichever indices are currently visible
47 * (the classic "recycler" pattern), rather than one Widget per item. Once a
48 * provider is set it takes over completely - addChild()'d children are
49 * ignored while one is active. Caution: a continuous gesture (drag/scroll)
50 * latches the Widget* returned by `at()` at kBegan and keeps calling
51 * methods on it through kChanged/kEnded - don't let the pool reassign that
52 * same slot to a different index while a gesture is still in progress.
53 */

80 /// Fast path for contentHeight(): when every item is exactly `height`
81 /// tall (uniform-height rows, the common case for a huge provider-backed
82 /// list), content height becomes `effectiveCount() * height` - an O(1)
83 /// calculation - instead of fetching and measuring every single item
84 /// every frame (see contentHeight()'s own doc comment on why that

86 /// measuring each item; irrelevant in addChild() vector mode, where

126 bool onGesture(const tinygpu::GestureEvent& event) override {
127 using tinygpu::GesturePhase;
128 using tinygpu::GestureType;
129
130 if (isContinuousType(event.type)) {
131 if (event.phase == GesturePhase::kBegan) {
132 // Only claim a pan/scroll for this container's own scroll if it
133 // actually has somewhere to scroll to - otherwise it would swallow
134 // every such gesture starting anywhere inside it (clamped to a
135 // no-op) before a nested Container/Screen underneath ever saw it,
136 // even one that genuinely needs to scroll. When this container has
137 // nothing to scroll, fall through to the same "find the hit child"
138 // path as any other continuous gesture, so - directly or through a
139 // few levels of nesting - whichever Container actually has
140 // overflowing content gets the drag. If more than one level in the
141 // chain needs to scroll, the outermost one still wins once it
142 // needs to (unchanged from before) - only the "nothing to scroll,
143 // so don't claim it" case is new.
144 if ((event.type == GestureType::kPan || event.type == GestureType::kScroll) &&
145 contentHeight() > bounds.h) {
146 scrollDragActive_ = true;
147 activeChild_ = nullptr;
148 } else {
149 scrollDragActive_ = false;
150 activeChild_ = hitTest(event.startPoint.x, event.startPoint.y);
151 }
152 }
153
154 bool handled = false;
155 if (scrollDragActive_) {
156 scrollOffset_ -= event.stepDeltaY;
157 clampScroll();
158 handled = true;
159 } else if (activeChild_ != nullptr && activeChild_->enabled) {
160 handled = dispatch(*activeChild_, event);
161 }
162
163 if (event.phase == GesturePhase::kEnded) {
164 activeChild_ = nullptr;
165 scrollDragActive_ = false;
166 }
167 return handled;
168 }
169
170 Widget<RGB_T>* hit = hitTest(event.point.x, event.point.y);
171 if (hit != nullptr && hit->enabled) return dispatch(*hit, event);
172 return false;
173 }

182 /// True if there's a child at (x, y) - this container's own coordinate
183 /// space - that's draggable, recursing into it via its own
184 /// isDraggableAt() if it's itself a composite (a nested Container,
185 /// Dialog, ...) - see Widget::isDraggableAt(). Wire Screen::
186 /// isDraggableAt() (or, for a Container nested inside another, this
187 /// same method on the outer one) to this so a drag starting on it, at
188 /// any nesting depth, is classified correctly.
189 bool isDraggableAt(int32_t x, int32_t y) const override {
190 Widget<RGB_T>* hit = hitTest(x, y);
191 if (hit == nullptr) return false;

195 /// True for a gesture type that spans multiple events (kBegan/kChanged/
196 /// kEnded) rather than firing once - used by callers that compose a
197 /// Container (Screen; Drawer/BottomSheet/Menu's own item area) to decide
198 /// whether to route a gesture here at all before it's mid-flight.
199 static bool isContinuousType(tinygpu::GestureType type) {
200 using tinygpu::GestureType;
201 switch (type) {
202 case GestureType::kDrag:
203 case GestureType::kPan:
204 case GestureType::kScroll:
205 case GestureType::kPinchIn:
206 case GestureType::kPinchOut:
207 case GestureType::kRotate:
208 return true;
209 default:
210 return false;
211 }

217 /// Height of this container's content, measured from its own bounds.y -
218 /// mirrors Screen::contentHeight() but relative to this container's own
219 /// top rather than the whole surface. Protected: Screen reuses this
220 /// directly for its own drawDirect()/drawScrollbarDirect(), since it's a
221 /// Container itself and drives the same scroll math one layer further out.
222 /// In provider mode this visits every logical index to find the tallest -
223 /// still O(count) rather than O(memory), but a huge count makes this (and
224 /// so every clampScroll() call, i.e. every frame) proportionally slower;
225 /// keep item heights cheap to compute if count is large, or call
226 /// setUniformItemHeight() for an O(1) alternative when rows are all the
227 /// same height.
228 int32_t contentHeight() const {
229 const int count = effectiveCount();
230 if (uniformItemHeight_ > 0) return count * uniformItemHeight_;
231 int32_t bottom = bounds.y;
232 for (int i = 0; i < count; ++i) {
233 Widget<RGB_T>* child = effectiveChild(i);
234 if (child != nullptr) bottom = std::max(bottom, child->bounds.bottom());
235 }
236 return bottom - bounds.y;
237 }

239 void clampScroll() {
240 const int32_t maxScroll = std::max(int32_t{0}, contentHeight() - bounds.h);
241 scrollOffset_ = std::min(std::max(scrollOffset_, int32_t{0}), maxScroll);
242 }

244 /// Number of children right now - the provider's count() if
245 /// setChildProvider() is active, else the addChild()'d vector's size.
246 /// Protected: Screen's drawDirect() drives the same loop one layer out.
247 int effectiveCount() const {
248 return childCountFn_ ? childCountFn_() : static_cast<int>(children_.size());
249 }

251 /// The child at `index` right now - from the provider if active (themed
252 /// immediately, since there's no fixed set to cascade setTheme() to ahead
253 /// of time - see setChildProvider()), else from the addChild()'d vector.
254 /// Protected: Screen's drawDirect() drives the same loop one layer out.
255 Widget<RGB_T>* effectiveChild(int index) const {
256 if (childAtFn_) {
257 Widget<RGB_T>& child = childAtFn_(index);
258 if (this->theme_ != nullptr) child.setTheme(*this->theme_);
259 return &child;
260 }
261 return children_[static_cast<size_t>(index)];
262 }

273 /// True if screen-space `b` overlaps this container's own bounds at all -
274 /// a cheap pre-filter so a child scrolled entirely out of view is skipped
275 /// without even being drawn (drawChildren()'s clip rect handles correctly
276 /// cropping the ones that only partially overlap).
277 bool intersectsBounds(const Bounds& b) const {
278 return b.bottom() > bounds.y && b.y < bounds.bottom() && b.right() > bounds.x && b.x < bounds.right();
279 }

281 /// Draws every visible child, each cropped to this container's own
282 /// bounds (see ISurface::pushClipRect()) - a child only partially
283 /// overlapping (scrolled halfway past the top/bottom edge) is cropped
284 /// to the visible portion instead of drawn in full or skipped outright.
285 void drawChildren(tinygpu::ISurface<RGB_T>& target) {
286 target.pushClipRect(toPx(bounds.x), toPx(bounds.y), toPx(bounds.w), toPx(bounds.h));
287 const int count = effectiveCount();
288 for (int i = 0; i < count; ++i) {
289 Widget<RGB_T>* child = effectiveChild(i);
290 if (child == nullptr || !child->visible) continue;
291 child->bounds.y -= scrollOffset_;
292 if (intersectsBounds(child->bounds)) child->draw(target);
293 child->bounds.y += scrollOffset_;
294 }
295 target.popClipRect();
296 }

298 /// A thin right-edge indicator of scroll position/range, confined to this
299 /// container's own bounds - mirrors Screen::drawScrollbar(). Omitted
300 /// entirely once content already fits.
301 void drawScrollbar(tinygpu::ISurface<RGB_T>& target) {
302 const int32_t content = contentHeight();
303 if (content <= bounds.h) return;
304
305 constexpr int32_t kBarWidth = 4;
306 const int32_t trackX = bounds.right() - kBarWidth;
307 target.fillRect(toPx(trackX), toPx(bounds.y), toPx(kBarWidth), toPx(bounds.h),
308 blend(this->theme().colors.surface, this->theme().colors.outline, 0.25f));
309
310 const int32_t maxScroll = content - bounds.h;
311 const int32_t thumbHeight = std::max(int32_t(24), bounds.h * bounds.h / content);
312 const int32_t thumbY =
313 maxScroll > 0 ? bounds.y + (scrollOffset_ * (bounds.h - thumbHeight)) / maxScroll : bounds.y;
314 target.fillRoundRect(toPx(trackX), toPx(thumbY), toPx(kBarWidth), toPx(thumbHeight), toPx(kBarWidth / 2),
315 this->theme().colors.primary);
316 }

318 /// Last-added child wins ties, i.e. later addChild() calls are treated as
319 /// drawn on top (matches drawChildren()). (x, y) is this container's own
320 /// coordinate space; translated into content space by the current scroll
321 /// offset before testing, since child bounds are authored in content
322 /// space (see the class comment).
323 Widget<RGB_T>* hitTest(int32_t x, int32_t y) const {
324 const int32_t contentY = y + scrollOffset_;
325 for (int i = effectiveCount() - 1; i >= 0; --i) {
326 Widget<RGB_T>* child = effectiveChild(i);
327 if (child != nullptr && child->visible && child->enabled && child->bounds.contains(x, contentY)) {
328 return child;
329 }
330 }
331 return nullptr;
332 }
333
334 /// Forwards `event` to `child`, translating its point/startPoint back into
335 /// content space first - mirrors Screen::dispatch().
336 bool dispatch(Widget<RGB_T>& child, const tinygpu::GestureEvent& event) {
337 if (scrollOffset_ == 0) return child.onGesture(event);
338 tinygpu::GestureEvent shifted = event;
339 shifted.point.y = static_cast<int16_t>(shifted.point.y + scrollOffset_);
340 shifted.startPoint.y = static_cast<int16_t>(shifted.startPoint.y + scrollOffset_);
341 return child.onGesture(shifted);
342 }
