17 * @brief Material Design button: Filled / Tonal / Outlined / Text / Elevated.
18 *
19 * Tap feedback is the (optional) ripple grown from the tap point - real
20 * touch drivers only report a recognized, completed tap (GestureType::kTap),
21 * not a raw finger-down instant, so the ripple *is* the button's touch
22 * feedback rather than a separate immediate "pressed" shade.
23 */

43 bool paintOutline = false;

46 case ButtonVariant::kFilled:
47 background = this->theme().colors.primary;
48 foreground = this->theme().colors.onPrimary;
49 break;
50 case ButtonVariant::kTonal:
51 background = this->theme().colors.secondaryContainer;
52 foreground = this->theme().colors.onSecondaryContainer;
53 break;
54 case ButtonVariant::kElevated:
55 background = this->theme().colors.surface;
56 foreground = this->theme().colors.primary;
57 break;
58 case ButtonVariant::kOutlined:
59 background = this->theme().colors.surface;
60 foreground = this->theme().colors.primary;

62 break;
63 case ButtonVariant::kText:

68 break;

81 toPx(this->bounds.w), toPx(this->bounds.h), toPx(radius),

111 bool onGesture(const tinygpu::GestureEvent& event) override {
112 if (!isTapGesture(event.type)) return false;
113 rippleOriginX_ = event.point.x;
114 rippleOriginY_ = event.point.y;
115 // lastUpdateMs_ is one frame stale (update() hasn't run yet this frame),
116 // which is imperceptible next to a ~260ms ripple.
117 rippleStartMs_ = lastUpdateMs_;
118 rippleActive_ = true;
119 if (onClick) onClick();
120 return true;
121 }
122
123 bool update(uint32_t nowMs) override {
124 const bool wasActive = rippleActive_;
125 lastUpdateMs_ = nowMs;
126 if (rippleActive_ && (nowMs - rippleStartMs_) >= kRippleDurationMs) {
127 rippleActive_ = false;
128 }
129 // The ripple's radius/opacity are a function of elapsed time recomputed
130 // in drawRipple() every frame, so any frame where it started the call
131 // active needs a redraw - including the final frame where it just
132 // finished fading out.
133 return wasActive;
134 }

136 private:
137 static constexpr uint32_t kRippleDurationMs = 260;
138
139 std::string label_;
140 ButtonVariant variant_ = ButtonVariant::kFilled;

142 bool rippleActive_ = false;
143 uint32_t rippleStartMs_ = 0;
144 uint32_t lastUpdateMs_ = 0;
145 int32_t rippleOriginX_ = 0;
146 int32_t rippleOriginY_ = 0;
147
148 /// Fills a circle growing from the tap point and fading out, clipped to
149 /// this button's rectangular bounds (a reasonable approximation - it
150 /// doesn't follow the pill's rounded corners, which isn't visible at
151 /// typical button sizes since the ripple fades out well before reaching
152 /// them).
153 void drawRipple(tinygpu::ISurface<RGB_T>& target, RGB_T background, RGB_T foreground) {
154 const uint32_t elapsed = lastUpdateMs_ - rippleStartMs_;
155 const float t = std::min(1.0f, static_cast<float>(elapsed) / kRippleDurationMs);
156 const float maxRadius =
157 static_cast<float>(this->bounds.w > this->bounds.h ? this->bounds.w : this->bounds.h);
158 const float radius = t * maxRadius;
159 const RGB_T rippleColor = blend(background, foreground, 0.20f * (1.0f - t));
160
161 const int32_t left = std::max(this->bounds.x, static_cast<int32_t>(rippleOriginX_ - radius));
162 const int32_t right = std::min(this->bounds.right(), static_cast<int32_t>(rippleOriginX_ + radius) + 1);
163 const int32_t top = std::max(this->bounds.y, static_cast<int32_t>(rippleOriginY_ - radius));
164 const int32_t bottom = std::min(this->bounds.bottom(), static_cast<int32_t>(rippleOriginY_ + radius) + 1);
165 const float radiusSq = radius * radius;
166
167 for (int32_t y = top; y < bottom; ++y) {
168 for (int32_t x = left; x < right; ++x) {
169 const float dx = static_cast<float>(x - rippleOriginX_);
170 const float dy = static_cast<float>(y - rippleOriginY_);
171 if (dx * dx + dy * dy <= radiusSq) {
172 target.setPixel(toPx(x), toPx(y), rippleColor);
173 }
174 }
175 }
176 }
177};
