13/**
14 * @brief Single-line text input box.
15 *
16 * TextField only holds and displays text and focus state - it has no idea
17 * how to turn touches into characters. Pair it with a Keyboard (Keyboard.h):
18 *
19 * Keyboard<RGB565> keyboard(Bounds(0, 420, 340, 190));
20 * TextField<RGB565> nameField(Bounds(16, 88, 220, 48), "Name");
21 * keyboard.manage(nameField); // wires nameField's focus to this keyboard
22 * screen.addWidget(nameField);
23 * screen.addWidget(keyboard); // add last so it draws on top when shown
24 *
25 * insertChar()/backspace()/handleEnter() (TextInputTarget, see Core/
26 * TextInputTarget.h) are what let a Keyboard - or any other input source -
27 * drive a focused field without TextField needing to know about it. Enter
28 * submits (see TextArea.h for the multi-line counterpart, where Enter
29 * inserts a newline instead).
30 *
31 * Bounds is split internally into a small label strip (top) and the input
32 * box itself (the rest) - both are part of this widget's own bounds, so a
33 * tap anywhere in either area focuses the field.
34 *
35 * Known limitation: text longer than the box is drawn past its right edge
36 * rather than scrolling/clipping - TinyGPU's fonts have no sub-rect clip,
37 * only surface-edge clipping. Fine for short inputs (names, search terms,
38 * ...); not meant for long free-form text.
39 */

148 void draw(tinygpu::ISurface<RGB_T>& target) override {
149 const int32_t labelHeight = hasLabel_ ? 14 : 0;
150 const int32_t errorHeight = error_.empty() ? 0 : 14;
151 const Bounds box(this->bounds.x, this->bounds.y + labelHeight, this->bounds.w,
152 this->bounds.h - labelHeight - errorHeight);
153 const RGB_T outside = onBackground_ ? this->theme().colors.background : this->theme().colors.surface;
154 const bool error = !error_.empty();
155
156 if (hasLabel_) {
157 RGB_T labelColor = error ? this->theme().colors.error
158 : focused_ ? this->theme().colors.primary
159 : this->theme().colors.onSurfaceVariant;
160 if (!this->enabled) labelColor = blend(labelColor, this->theme().colors.surface, 0.5f);
161 this->theme().typography.label->drawText(target, static_cast<int16_t>(this->bounds.x),
162 static_cast<int16_t>(this->bounds.y), label_.c_str(),
163 labelColor, outside, false);
164 }
165 if (error) {
166 this->theme().typography.label->drawText(target, static_cast<int16_t>(this->bounds.x + 4),
167 static_cast<int16_t>(box.bottom() + 3), error_.c_str(),
168 this->theme().colors.error, outside, false);
169 }
170
171 RGB_T borderColor = error ? this->theme().colors.error
172 : focused_ ? this->theme().colors.primary
173 : this->theme().colors.outline;
174 if (!this->enabled) borderColor = blend(borderColor, this->theme().colors.surface, 0.5f);
175 const size_t radius = toPx(this->theme().shape.small);
176 target.fillRoundRect(toPx(box.x), toPx(box.y), toPx(box.w), toPx(box.h), radius,
177 this->theme().colors.surface);
178 target.drawRoundRect(toPx(box.x), toPx(box.y), toPx(box.w), toPx(box.h), radius, borderColor);
179
180 const int32_t pad = 8;
181 const int32_t textX = box.x + pad;
182 const int32_t textY = box.centerY() - static_cast<int32_t>(this->theme().typography.body->getHeight(1)) / 2;
183
184 if (text_.empty() && !focused_) {
185 if (!placeholder_.empty()) {
186 this->theme().typography.body->drawText(target, static_cast<int16_t>(textX),
187 static_cast<int16_t>(textY), placeholder_.c_str(),
188 this->theme().colors.onSurfaceVariant, this->theme().colors.surface, false);
189 }
190 return;
191 }
192
193 const std::string shown = password_ ? std::string(text_.size(), '*') : text_;
194 this->theme().typography.body->drawText(target, static_cast<int16_t>(textX),
195 static_cast<int16_t>(textY), shown.c_str(),
196 this->theme().colors.onSurface, this->theme().colors.surface, false);
197
198 if (focused_ && cursorVisible_) {
199 const size_t textWidth = this->theme().typography.body->measureTextWidth(shown.c_str());
200 const int32_t cursorX = textX + static_cast<int32_t>(textWidth) + 1;
201 target.drawLine(toPx(cursorX), toPx(box.y + 6), toPx(cursorX), toPx(box.bottom() - 6),
202 this->theme().colors.primary);
203 }
204 }

212 bool update(uint32_t nowMs) override {
213 const bool wasVisible = cursorVisible_;
214 if (!focused_) {

218 if (nowMs - lastBlinkMs_ >= kBlinkIntervalMs) {

221 }

223 }
