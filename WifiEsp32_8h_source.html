42 bool startScan() override {
43 // a connection attempt blocks the scan: stop it
44 if (connecting_ && WiFi.status() != WL_CONNECTED) {
45 WiFi.disconnect();
46 connecting_ = false;
47 }
48 return WiFi.scanNetworks(true /* async */, false /* hidden */) != WIFI_SCAN_FAILED;

51 bool scanDone(std::vector<WifiNetwork>& out) override {
52 const int n = WiFi.scanComplete();
53 if (n == WIFI_SCAN_RUNNING) return false;
54 out.clear();
55 for (int i = 0; i < n; ++i) {
56 WifiNetwork net;
57 net.ssid = WiFi.SSID(i).c_str();
58 net.rssi = WiFi.RSSI(i);
59 net.secure = WiFi.encryptionType(i) != WIFI_AUTH_OPEN;
60 out.push_back(net);
61 }
62 if (n >= 0) WiFi.scanDelete();
63 normalizeNetworks(out);
64 return true;
65 }
