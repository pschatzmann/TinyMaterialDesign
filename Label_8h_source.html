14/// Maps a TypographyRole to the theme's corresponding font - shared by
15/// every widget that lets a caller pick a text size (Label, ListItem, ...)
16/// rather than each duplicating this switch.
17template <typename RGB_T>
18tinygpu::IFont<RGB_T>* fontForTypographyRole(TypographyRole role, const MaterialTheme<RGB_T>& theme) {
19 switch (role) {
20 case TypographyRole::kHeadline:
21 return theme.typography.headline;
22 case TypographyRole::kTitle:
23 return theme.typography.title;
24 case TypographyRole::kLabel:
25 return theme.typography.label;
26 case TypographyRole::kBody:
27 default:
28 return theme.typography.body;
29 }
30}
