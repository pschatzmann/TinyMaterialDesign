31 bool isSelected(int index) const { return (selectedMask_ & (1u << index)) != 0; }
32 void setSelected(int index, bool selected) {
33 if (index < 0 || index >= static_cast<int>(segments_.size())) return;
34 if (!multiSelect_) {

89 const int count = static_cast<int>(segments_.size());
90 if (!isTapGesture(event.type) || count == 0) return false;
91 const int32_t segmentWidth = this->bounds.w / count;
92 int index = (event.point.x - this->bounds.x) / segmentWidth;
93 if (index < 0 || index >= count) return true;
94
95 if (multiSelect_) {
96 setSelected(index, !isSelected(index));
97 } else {

99 }
