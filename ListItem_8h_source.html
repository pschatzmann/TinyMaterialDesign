14/**
15 * @brief List row after the Material 3 list spec: a leading avatar or icon,
16 * a headline, an optional supporting text, a trailing text or icon, and an
17 * optional divider. Selected rows draw as a filled pill (the navigation-drawer
18 * look). Everything beyond the headline is opt-in, so a plain ListItem looks
19 * as before.
20 *
21 * ListItem<RGB565> row(Bounds(0, y, 240, ListItem<RGB565>::kOneLineHeight), "Quarterly report");
22 * row.setSupportingText("Alice <alice@example.com>"); // two-line item
23 * row.setLeadingText("A"); // 40 dp avatar with a letter
24 * row.setTrailingText("2 files"); // label on the right
25 * row.setDivider(true); // divider at the bottom edge
26 *
27 * Sizes (dp, the same as px here): one-line item 56, two-line item 72, avatar
28 * 40, icon 24, dividers 1 with an inset of 16 from the text.
29 */
30template <typename RGB_T = TINYMD_DEFAULT_RGB_T>

109 void draw(tinygpu::ISurface<RGB_T>& target) override {
110 const MaterialTheme<RGB_T>& t = this->theme();
111 RGB_T background = selected_ ? t.colors.secondaryContainer : t.colors.surface;
112 RGB_T foreground = selected_ ? t.colors.onSecondaryContainer : t.colors.onSurfaceVariant;
113 RGB_T textColor = selected_ ? t.colors.onSecondaryContainer : t.colors.onSurface;
114 RGB_T outline = hasDividerColor_ ? dividerColor_ : t.colors.outline;
115 if (hasBackgroundColor_ && !selected_) {
116 background = backgroundColor_;
117 target.fillRoundRect(toPx(this->bounds.x), toPx(this->bounds.y), toPx(this->bounds.w),
118 toPx(this->bounds.h), toPx(t.shape.medium), backgroundColor_);
119 }
120 if (!this->enabled) {
121 background = blend(background, t.colors.surface, 0.5f);
122 foreground = blend(foreground, t.colors.surface, 0.5f);
123 textColor = blend(textColor, t.colors.surface, 0.5f);
124 outline = blend(outline, t.colors.surface, 0.5f);
125 }
126
127 const int32_t margin = 8;
128 const Bounds pill = this->bounds.inset(margin);
129
130 if (selected_) {
131 target.fillRoundRect(toPx(pill.x), toPx(pill.y), toPx(pill.w), toPx(pill.h),
132 toPx(t.shape.full), background);
133 }
134
135 const int32_t pad = 16;
136 const int32_t avatarSize = pill.h < 40 ? pill.h : 40; // 40 dp avatar
137 const int32_t iconSize = 24; // 24 dp icon
138 int32_t textX = pill.x + pad;
139
140 // ---- leading slot
141 if (!leadingText_.empty()) {
142 const int32_t r = avatarSize / 2;
143 const int32_t cx = pill.x + pad + r;
144 const int32_t cy = pill.centerY();
145 target.fillCircle(static_cast<size_t>(cx), static_cast<size_t>(cy), static_cast<size_t>(r),
146 t.colors.primaryContainer);
147 tinygpu::IFont<RGB_T>& letters = *fontForTypographyRole(TypographyRole::kLabel, t);
148 const int32_t lw = static_cast<int32_t>(letters.measureTextWidth(leadingText_.c_str()));
149 const int32_t lh = static_cast<int32_t>(letters.getHeight(1));
150 letters.drawText(target, static_cast<int16_t>(cx - lw / 2), static_cast<int16_t>(cy - lh / 2),
151 leadingText_.c_str(), t.colors.onPrimaryContainer, t.colors.primaryContainer, false);
152 textX = pill.x + pad + avatarSize + pad;
153 } else if (icon_ != nullptr) {
154 const Bounds iconRect(pill.x + pad, pill.centerY() - iconSize / 2, iconSize, iconSize);
155 icon_(target, iconRect, foreground, /*thickness=*/2);
156 textX = iconRect.right() + pad;
157 }
158
159 // ---- trailing slot (drawn first so the headline can stop before it)
160 int32_t textRight = pill.right() - pad;
161 if (!trailingText_.empty()) {
162 tinygpu::IFont<RGB_T>& small = *fontForTypographyRole(TypographyRole::kLabel, t);
163 const int32_t w = static_cast<int32_t>(small.measureTextWidth(trailingText_.c_str()));
164 const int32_t h = static_cast<int32_t>(small.getHeight(1));
165 small.drawText(target, static_cast<int16_t>(textRight - w), static_cast<int16_t>(pill.centerY() - h / 2),
166 trailingText_.c_str(), foreground, background, false);
167 textRight -= w + pad;
168 }
169 if (trailingIcon_ != nullptr) {
170 const Bounds iconRect(textRight - iconSize, pill.centerY() - iconSize / 2, iconSize, iconSize);
171 trailingIcon_(target, iconRect, foreground, /*thickness=*/2);
172 textRight = iconRect.x - pad;
173 }
174
175 // ---- headline and supporting text, centred as one block, clipped to the space left
176 target.pushClipRect(toPx(textX), toPx(pill.y), toPx(textRight - textX), toPx(pill.h));
177 tinygpu::IFont<RGB_T>& font = titleFont_ != nullptr ? *titleFont_ : *fontForTypographyRole(role_, t);
178 const int32_t titleH = static_cast<int32_t>(font.getHeight(1));
179 const int32_t textWidth = textRight - textX;
180 if (supporting_.empty()) {
181 const int32_t textY = pill.centerY() - titleH / 2;
182 font.drawText(target, static_cast<int16_t>(textX), static_cast<int16_t>(textY), title_.c_str(),
183 titleColor(textColor), background, false);
184 } else {
185 tinygpu::IFont<RGB_T>& small = subtitleFont_ != nullptr ? *subtitleFont_ : *fontForTypographyRole(TypographyRole::kLabel, t);
186 const int32_t smallH = static_cast<int32_t>(small.getHeight(1));
187 const int32_t gap = 2;
188 const int32_t top = pill.centerY() - (titleH + gap + smallH) / 2;
189 font.drawText(target, static_cast<int16_t>(textX), static_cast<int16_t>(top), title_.c_str(), titleColor(textColor),
190 background, false);
191 (void)textWidth; // long supporting text is clipped by the list
192 small.drawText(target, static_cast<int16_t>(textX), static_cast<int16_t>(top + titleH + gap),
193 supporting_.c_str(), foreground, background, false);
194 }
195
196 target.popClipRect();
197
198 // ---- divider: full width, or inset from the text
199 if (divider_) {
200 const int32_t y = this->bounds.y + this->bounds.h - 1;
201 const int32_t x0 = dividerInset_ ? textX : this->bounds.x;
202 const int32_t x1 = this->bounds.x + this->bounds.w - (dividerInset_ ? pad : 0);
203 target.drawLine(static_cast<size_t>(x0), static_cast<size_t>(y), static_cast<size_t>(x1),
204 static_cast<size_t>(y), outline);
205 }
206 }
