14/**
15 * @brief Modal panel sliding up from the bottom edge, full width, with
16 * rounded top corners and a drag-handle bar - Material's "modal bottom
17 * sheet". Holds a list of items (typically ListItem or Button) the same
18 * non-owning way Drawer holds its items, in a Container (Core/Container.h)
19 * that scrolls automatically once they overflow the sheet's height.
20 *
21 * `bounds` is the sheet's own rect (e.g. Bounds(0, screenH - 220, screenW,
22 * 220)) - not the full screen; the scrim is drawn separately covering the
23 * whole target surface, same convention as Dialog/Drawer. Show it with
24 * Screen::presentDialog(sheet) and dismiss it from an item's onClick or
25 * from onScrimTap.
26 *
27 * Known limitation: same as Drawer's - Screen::drawDirect()'s per-widget-
28 * buffer rendering path draws items at their unscrolled position; only the
29 * regular draw() path scrolls correctly. See Drawer.h's class comment.
30 */

80 /// Scrim + panel fill + handle + title, no items - see
81 /// Widget::drawBackground() and childCount()/child() (same rationale as
82 /// Drawer's own).
83 void drawBackground(tinygpu::ISurface<RGB_T>& target) override {
84 drawScrim(target);
85
86 const size_t radius = toPx(this->theme().shape.large);
87 // Rounded top corners only: draw a fully-rounded rect, then square off
88 // the bottom half by overpainting it flat - cheaper than a 4-independent-
89 // corner primitive, which ISurface doesn't expose.
90 target.fillRoundRect(toPx(this->bounds.x), toPx(this->bounds.y), toPx(this->bounds.w),
91 toPx(this->bounds.h), radius, this->theme().colors.surface);
92 const int32_t squareTop = this->bounds.y + static_cast<int32_t>(radius);
93 if (squareTop < this->bounds.bottom()) {
94 target.fillRect(toPx(this->bounds.x), toPx(squareTop), toPx(this->bounds.w),
95 toPx(this->bounds.bottom() - squareTop), this->theme().colors.surface);
96 }
97
98 constexpr int32_t kHandleWidth = 32;
99 constexpr int32_t kHandleHeight = 4;
100 const int32_t handleY = this->bounds.y + 8;
101 target.fillRoundRect(toPx(this->bounds.centerX() - kHandleWidth / 2), toPx(handleY),
102 toPx(kHandleWidth), toPx(kHandleHeight), toPx(kHandleHeight / 2),
103 this->theme().colors.outline);
104
105 if (!title_.empty()) {
106 tinygpu::IFont<RGB_T>& font = *this->theme().typography.title;
107 const int32_t pad = this->theme().spacing;
108 font.drawText(target, static_cast<int16_t>(this->bounds.x + pad * 2),
109 static_cast<int16_t>(handleY + kHandleHeight + pad), title_.c_str(),
110 this->theme().colors.onSurface, this->theme().colors.surface, false);
111 }
112 }

117 /// A continuous drag starting over the item list scrolls it (see
118 /// Container::onGesture()); a tap on an item is forwarded to it; a tap
119 /// outside the panel fires onScrimTap; everything else is swallowed.
120 bool onGesture(const tinygpu::GestureEvent& event) override {
121 syncItemsBounds();
122
123 if (Container<RGB_T>::isContinuousType(event.type)) {
124 if (event.phase == tinygpu::GesturePhase::kBegan) {
125 routingToItems_ = items_.bounds.contains(event.startPoint.x, event.startPoint.y);
126 }
127 return routingToItems_ ? items_.onGesture(event) : true;
128 }
129
130 if (items_.bounds.contains(event.point.x, event.point.y) && items_.onGesture(event)) {
131 return true;
132 }
133 if (isTapGesture(event.type) && !this->bounds.contains(event.point.x, event.point.y)) {
134 if (onScrimTap) onScrimTap();
135 }
136 return true;
137 }

139 bool update(uint32_t nowMs) override { return items_.update(nowMs); }

143 bool isDraggableAt(int32_t x, int32_t y) const override {

145 }
146
147 private:
148 int32_t contentTop() const {
149 constexpr int32_t kHandleHeight = 4;
150 constexpr int32_t kPad = 8;
151 // Approximates the title typography's line height without touching
152 // theme() (see itemRect()'s comment) - close enough for row layout, and
153 // drawBackground() still measures the real font for the title itself.
154 constexpr int32_t kApproxTitleHeight = 20;
155 const int32_t afterHandle = this->bounds.y + 8 + kHandleHeight + kPad;
156 if (title_.empty()) return afterHandle;
157 return afterHandle + kApproxTitleHeight + kPad;
158 }
159
160 /// Keeps the item Container's own bounds in sync with this sheet's -
161 /// items start below the handle/title, not at bounds.y itself.
162 void syncItemsBounds() {
163 const int32_t top = contentTop();
164 items_.bounds = Bounds(this->bounds.x, top, this->bounds.w, this->bounds.bottom() - top);
165 }
166
167 std::string title_;
168 Container<RGB_T> items_;
169 bool routingToItems_ = false;
170};
