13/// Per-channel linear blend between two colors (t=0 -> a, t=1 -> b), used
14/// for pressed/disabled state shading, cheap elevation "shadows", and the
15/// ripple fade. RGB_T has no native alpha channel, so this is the cheapest
16/// approximation that still looks right at typical embedded-UI sizes.
17template <typename RGB_T>
18RGB_T blend(RGB_T a, RGB_T b, float t) {
19 t = std::min(1.0f, std::max(0.0f, t));
20 auto lerp = [t](uint8_t x, uint8_t y) {
21 return static_cast<uint8_t>(x + (static_cast<float>(y) - x) * t);
22 };
23 return RGB_T(lerp(a.getRed(), b.getRed()), lerp(a.getGreen(), b.getGreen()),
24 lerp(a.getBlue(), b.getBlue()));
25}

47/// @brief Bundles a color scheme, shape tokens, typography, and spacing into

61/// Function-local static fonts, constructed once on first use - the same
62/// "kept as static instances" idiom TinyGPU's own basic-example.ino uses,
63/// just centralized here instead of left to each sketch.
64template <typename RGB_T>
65Typography<RGB_T> defaultTypography() {
66 static tinygpu::Font5x7<RGB_T> label;
67 static tinygpu::Font8x8<RGB_T> body;
68 static tinygpu::Font12x12<RGB_T> title;
69 static tinygpu::Font16x24<RGB_T> headline;
70 return Typography<RGB_T>{&label, &body, &title, &headline};
71}
