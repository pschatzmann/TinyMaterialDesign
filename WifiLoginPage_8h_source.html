67/**
68 * @brief WiFi selection and login as a Page: the state of the connection,
69 * SSID and password (masked, "Show password"), Connect / Scan, the networks
70 * of the scan (tap one to select it: open networks connect at once, for
71 * secured ones the keyboard opens for the password) and "Continue without
72 * WiFi".
73 *
74 * The WiFi itself is a WifiControl: Esp32Wifi (Wifi/WifiEsp32.h) on the
75 * ESP32, SimulatedWifi on the desktop. Show the page on its own, or add it
76 * to a PageNavigator; call tick() regularly while it is shown (the
77 * PageNavigator does that) - it polls the scan and the connection and
78 * returns true when the page must be redrawn.
79 *

81 * WifiLoginPage<RGB565> login(keyboard);

91 * login.onShow();

202 /// Polls the scan and the connection: true if the page changed.
203 bool tick(uint32_t nowMs) override {
204 if (wifi_ == nullptr || nowMs - lastTickMs_ < 250) return false;
205 lastTickMs_ = nowMs;
206 bool changed = false;
207 if (scanning_ && wifi_->scanDone(networks_)) {
208 scanning_ = false;
209 if (networks_.size() > kMaxNetworks) networks_.resize(kMaxNetworks);
210 changed = true;
211 }
212 const WifiControl::State state = wifi_->state();
213 if (state != lastState_) {
214 if (state == WifiControl::State::kConnected && startup_ && lastState_ == WifiControl::State::kConnecting) {
215 doneAtMs_ = nowMs + 1500; // show "connected" briefly
216 }
217 lastState_ = state;
218 changed = true;
219 }
220 if (doneAtMs_ != 0 && static_cast<int32_t>(nowMs - doneAtMs_) >= 0) {
221 doneAtMs_ = 0;
222 startup_ = false;
223 if (onDone) onDone();
224 }
225 if (changed) {
226 layout(); // highlights the connected network
227 refreshStatus();
228 }
229 return changed;
230 }

271 void refreshStatus() {
272 if (!built_) return;
273 if (wifi_ == nullptr) {
274 status_.setText("No WiFi available");
275 return;
276 }
277 const ColorScheme<RGB_T>& c = this->theme().colors;
278 char buf[96];
279 switch (wifi_->state()) {
280 case WifiControl::State::kConnected:
281 snprintf(buf, sizeof(buf), "Connected to %s (%s)", wifi_->ssid().c_str(), wifi_->ip().c_str());
282 status_.setColor(c.primary);
283 break;
284 case WifiControl::State::kConnecting:
285 snprintf(buf, sizeof(buf), "Connecting to %s ...", wifi_->ssid().c_str());
286 status_.setColor(c.onSurfaceVariant);
287 break;
288 case WifiControl::State::kFailed:
289 snprintf(buf, sizeof(buf), "Could not connect to %s: check the password", wifi_->ssid().c_str());
290 status_.setColor(c.error);
291 break;
292 default:
293 snprintf(buf, sizeof(buf), "Not connected: select a network");
294 status_.setColor(c.onSurfaceVariant);
