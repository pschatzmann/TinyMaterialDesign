72/**
73 * @brief WiFi selection and login as a Page: the state of the connection,
74 * the SSID with a button that opens the networks of a scan as pop-up list,
75 * the password (masked, "Show password"), Connect (Disconnect while
76 * connected) and Cancel (leaves the page without connecting).
77 *
78 * - When a field is tapped, the page scrolls it to the top, so that it stays
79 * visible above the keyboard.
80 * - Connect scrolls back to the top: the state of the connection is visible.
81 * - Selecting a network in the pop-up connects at once to an open network
82 * and opens the keyboard for the password of a secured one.
83 * - Once a connection works, the page calls onDone (after showing
84 * "Connected" briefly).
85 *
86 * The WiFi itself is a WifiControl: Esp32Wifi (Wifi/WifiEsp32.h) on the
87 * ESP32, SimulatedWifi on the desktop. Show the page on its own, or add it
88 * to a PageNavigator; call tick() regularly while it is shown (the
89 * PageNavigator does that) - it polls the scan and the connection and
90 * returns true when the page must be redrawn.
91 *
92 * Keyboard<RGB565> keyboard(Bounds(0, 150, 240, 170));
93 * WifiLoginPage<RGB565> login(keyboard, app.screen());
94 * Esp32Wifi wifi;
95 *
96 * void setup() {
97 * app.begin();
98 * wifi.begin();
99 * login.setWifi(&wifi);
100 * login.bounds = Bounds(0, 40, 240, 280);
101 * app.screen().addWidget(login);
102 * login.build();
103 * login.onShow();
104 * app.screen().addFixedWidget(keyboard);
105 * login.onDone = []() { ... }; // connected or cancelled
106 * }
107 * void loop() {
108 * if (login.tick(millis())) app.screen().invalidate();
109 * app.update();
110 * }

232 /// Polls the scan and the connection: true if the page changed.
233 bool tick(uint32_t nowMs) override {
234 if (wifi_ == nullptr || nowMs - lastTickMs_ < 250) return false;
235 lastTickMs_ = nowMs;
236 bool changed = false;
237 if (scanning_ && wifi_->scanDone(networks_)) {
238 scanning_ = false;
239 if (networks_.size() > kMaxNetworks) networks_.resize(kMaxNetworks);
240 if (popupOpen_) fillPopup();
241 changed = true;
242 }
243 const WifiControl::State state = wifi_->state();
244 if (state != lastState_) {
245 // a new connection: show "Connected" briefly, then close
246 if (state == WifiControl::State::kConnected) doneAtMs_ = nowMs + 1500;
247 lastState_ = state;
248 changed = true;
249 }
250 if (doneAtMs_ != 0 && static_cast<int32_t>(nowMs - doneAtMs_) >= 0) {
251 doneAtMs_ = 0;
252 if (onDone) onDone();
253 }
254 if (changed) refreshStatus();
255 return changed;
256 }

354 void refreshStatus() {
355 if (!built_) return;
356 if (wifi_ == nullptr) {
357 status_.setText("No WiFi available");
358 return;
359 }
360 const ColorScheme<RGB_T>& c = this->theme().colors;
361 char buf[96];
362 switch (wifi_->state()) {
363 case WifiControl::State::kConnected:
364 snprintf(buf, sizeof(buf), "Connected to %s (%s)", wifi_->ssid().c_str(), wifi_->ip().c_str());
365 status_.setColor(c.primary);
366 break;
367 case WifiControl::State::kConnecting:
368 snprintf(buf, sizeof(buf), "Connecting to %s ...", wifi_->ssid().c_str());
369 status_.setColor(c.onSurfaceVariant);
370 break;
371 case WifiControl::State::kFailed:
372 snprintf(buf, sizeof(buf), "Could not connect to %s: check the password", wifi_->ssid().c_str());
373 status_.setColor(c.error);
374 break;
375 default:
376 snprintf(buf, sizeof(buf), scanning_ ? "Not connected - scanning for networks ..."
377 : "Not connected: select a network");
378 status_.setColor(c.onSurfaceVariant);
379 break;
380 }
381 status_.setText(buf);
382 connect_.setLabel(connected() ? "Disconnect" : "Connect");
383 screen_.invalidate();
384 }
