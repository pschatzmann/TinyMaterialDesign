35 void draw(tinygpu::ISurface<RGB_T>& target) override {
36 const int count = static_cast<int>(tabs_.size());
37 if (count == 0) return;
38 const LinearLayout slots(this->bounds, LayoutAxis::Horizontal, /*spacing=*/0);
39 tinygpu::IFont<RGB_T>& font = *this->theme().typography.label;
40
41 for (int i = 0; i < count; ++i) {
42 const Bounds slot = slots.itemRect(i, count);
43 const bool selected = i == selectedIndex_;
44 RGB_T textColor = selected ? this->theme().colors.primary : this->theme().colors.onSurfaceVariant;
45 if (!this->enabled) textColor = blend(textColor, this->theme().colors.surface, 0.5f);
46
47 const size_t textWidth = font.measureTextWidth(tabs_[i]);
48 const int32_t textX = slot.centerX() - static_cast<int32_t>(textWidth) / 2;
49 const int32_t textY = this->bounds.centerY() - static_cast<int32_t>(font.getHeight(1)) / 2;
50 font.drawText(target, static_cast<int16_t>(textX), static_cast<int16_t>(textY), tabs_[i],
51 textColor, this->theme().colors.surface, false);
52 }
53
54 constexpr int32_t kIndicatorHeight = 3;
55 const Bounds selectedSlot = slots.itemRect(selectedIndex_, count);
56 RGB_T indicatorColor =
57 this->enabled ? this->theme().colors.primary
58 : blend(this->theme().colors.primary, this->theme().colors.surface, 0.5f);
59 target.fillRect(toPx(selectedSlot.x), toPx(this->bounds.bottom() - kIndicatorHeight),
60 toPx(selectedSlot.w), toPx(kIndicatorHeight), indicatorColor);

63 bool onGesture(const tinygpu::GestureEvent& event) override {
64 const int count = static_cast<int>(tabs_.size());
65 if (!isTapGesture(event.type) || count == 0) return false;
66 const int32_t tabWidth = this->bounds.w / count;
67 int index = (event.point.x - this->bounds.x) / tabWidth;
68 if (index < 0) index = 0;
69 if (index >= count) index = count - 1;
70 if (index != selectedIndex_) {

72 if (onChange) onChange(selectedIndex_);
73 }
