32 /// Bounds for a `width` x `height` rect anchored at `anchor`, inset by margin.
33 Bounds rect(Anchor anchor, int32_t width, int32_t height) const {
34 int32_t x;
35 int32_t y;
36 switch (anchor) {
37 case Anchor::TopLeft:
38 x = container_.x + margin_;
39 y = container_.y + margin_;
40 break;
41 case Anchor::TopCenter:
42 x = container_.centerX() - width / 2;
43 y = container_.y + margin_;
44 break;
45 case Anchor::TopRight:
46 x = container_.right() - margin_ - width;
47 y = container_.y + margin_;
48 break;
49 case Anchor::CenterLeft:
50 x = container_.x + margin_;
51 y = container_.centerY() - height / 2;
52 break;
53 case Anchor::Center:
54 x = container_.centerX() - width / 2;
55 y = container_.centerY() - height / 2;
56 break;
57 case Anchor::CenterRight:
58 x = container_.right() - margin_ - width;
59 y = container_.centerY() - height / 2;
60 break;
61 case Anchor::BottomLeft:
62 x = container_.x + margin_;
63 y = container_.bottom() - margin_ - height;
64 break;
65 case Anchor::BottomCenter:
66 x = container_.centerX() - width / 2;
67 y = container_.bottom() - margin_ - height;
68 break;
69 case Anchor::BottomRight:
70 default:
71 x = container_.right() - margin_ - width;
72 y = container_.bottom() - margin_ - height;
73 break;
74 }
75 return Bounds(x, y, width, height);
76 }
