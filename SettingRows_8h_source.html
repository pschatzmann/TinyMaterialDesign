15/**
16 * @file SettingRows.h
17 * @brief Rows for settings and detail pages: a caption combined with a
18 * control or a value. Each row is one widget, so a page only places rows
19 * from top to bottom. By default they are drawn on the page background; call
20 * setOnSurface(true) for rows inside a Card.
21 *
22 * - SliderRow: caption and formatted value, a Slider below (kHeight 40)
23 * - SwitchRow: caption, Switch at the right; tapping the row toggles it
24 * - RadioRow: RadioButton and caption; tapping the row selects it
25 * - KeyValueRow: key at the left, value at the right
26 */
27
28/// Common part of the rows: caption and background.
29template <typename RGB_T = TINYMD_DEFAULT_RGB_T>
30class SettingRow : public Widget<RGB_T> {
31 public:
32 void setCaption(const char* caption) { caption_ = caption != nullptr ? caption : ""; }
33 const std::string& caption() const { return caption_; }
34 /// Draw on a surface (e.g. inside a Card) instead of the page background.
35 void setOnSurface(bool onSurface) { onSurface_ = onSurface; }
36
37 protected:
38 std::string caption_;
39 bool onSurface_ = false;
40
41 RGB_T background() const { return onSurface_ ? this->theme().colors.surface : this->theme().colors.background; }
42
43 void text(tinygpu::ISurface<RGB_T>& target, tinygpu::IFont<RGB_T>& font, int32_t x, int32_t y, const char* s,
44 RGB_T color) {
45 font.drawText(target, static_cast<int16_t>(x), static_cast<int16_t>(y), s, color, background(), false);
46 }
47
48 RGB_T dimmed(RGB_T color) const { return this->enabled ? color : blend(color, background(), 0.5f); }
49};

51/**
52 * @brief Caption and formatted value on one line, a Slider below. The value
53 * is quantized to `step`:
54 *
55 * SliderRow<RGB565> gain("Gain", 0, 40, 1, [](float v, char* b, size_t n) {
56 * snprintf(b, n, "%+.0f dB", v);
57 * });
58 * gain.onChange = [](float v) { ... };
59 */
60template <typename RGB_T = TINYMD_DEFAULT_RGB_T>
61class SliderRow : public SettingRow<RGB_T> {
62 public:
63 using Formatter = std::function<void(float value, char* buf, size_t len)>;
64 static constexpr int32_t kHeight = 40;
65
66 SliderRow() = default;
67 SliderRow(const char* caption, float minValue, float maxValue, float step, Formatter format = nullptr)
68 : step_(step), format_(std::move(format)), slider_(Bounds(), minValue, maxValue) {
69 this->setCaption(caption);
70 slider_.onChange = [this](float v) {
71 const float q = quantize(v);
72 if (q == value_) return;
73 value_ = q;
74 if (onChange) onChange(value_);
75 };
76 }
77
78 std::function<void(float)> onChange;
79
80 float value() const { return value_; }
81 void setValue(float value) {
82 value_ = quantize(value);
83 slider_.setValue(value_);
84 }
85
86 void setTheme(const MaterialTheme<RGB_T>& theme) override {
87 Widget<RGB_T>::setTheme(theme);
88 slider_.setTheme(theme);
89 }
90
91 void draw(tinygpu::ISurface<RGB_T>& target) override {
92 const MaterialTheme<RGB_T>& t = this->theme();
93 const Bounds& b = this->bounds;
94 tinygpu::IFont<RGB_T>& small = *t.typography.label;
95 tinygpu::IFont<RGB_T>& body = *t.typography.body;
96 this->text(target, small, b.x, b.y + 3, this->caption_.c_str(), this->dimmed(t.colors.onSurfaceVariant));
97 char buf[24] = "";
98 if (format_) {
99 format_(value_, buf, sizeof(buf));
100 } else {
101 snprintf(buf, sizeof(buf), "%g", value_);
102 }
103 this->text(target, body, b.right() - static_cast<int32_t>(body.measureTextWidth(buf)), b.y + 2, buf,
104 this->dimmed(t.colors.onSurface));
105 syncSlider();
106 slider_.enabled = this->enabled;
107 slider_.draw(target);
108 }
109
110 bool onGesture(const tinygpu::GestureEvent& event) override {
111 syncSlider();
112 return slider_.onGesture(event);
113 }
114
115 bool isDraggableAt(int32_t x, int32_t y) const override {
116 (void)x;
117 return y >= this->bounds.y + kTextHeight;
118 }
119
120 private:
121 static constexpr int32_t kTextHeight = 14;
122 float step_ = 1.0f;
123 float value_ = 0.0f;
124 Formatter format_;
125 Slider<RGB_T> slider_;
126
127 float quantize(float v) const { return step_ > 0 ? roundf(v / step_) * step_ : v; }
128
129 /// The slider band below the text (the bounds move while a Container scrolls).
130 void syncSlider() {
131 const Bounds& b = this->bounds;
132 slider_.bounds = Bounds(b.x + 9, b.y + kTextHeight + 2, b.w - 18, 18);
133 }
134};
