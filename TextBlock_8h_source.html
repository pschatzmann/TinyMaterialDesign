9/**
10 * @brief Multi-line text that is wrapped at word boundaries to the widget
11 * width ('\n' starts a new line) - for notes, explanations and messages.
12 * Unlike Label it can span several lines; heightFor() tells how much room
13 * the text needs, so a layout can size the widget to its text:
14 *
15 * TextBlock<RGB565> note(Bounds(8, y, 224, 0), "A long explanation ...");
16 * page.addChild(note); // gives it the theme
17 * note.bounds.h = note.heightFor(note.bounds.w);
18 *
19 * Default: label font, onSurfaceVariant color, drawn on the page background.
20 */

73 /// Calls fn(line) for each line of `text` wrapped to `width`.
74 template <typename Fn>
75 static void forEachLine(const std::string& text, int32_t width, tinygpu::IFont<RGB_T>& font, Fn fn) {
76 std::string line, word;
77 auto fits = [&](const std::string& s) { return static_cast<int32_t>(font.measureTextWidth(s.c_str())) <= width; };
78 for (size_t i = 0; i <= text.size(); ++i) {
79 const char c = i < text.size() ? text[i] : ' ';
80 if (c != ' ' && c != '\n') {
81 word += c;
82 continue;
83 }
84 if (!word.empty()) {
85 const std::string candidate = line.empty() ? word : line + " " + word;
86 if (!line.empty() && !fits(candidate)) {
87 fn(line);
88 line = word;
89 } else {
90 line = candidate;
91 }
92 word.clear();
93 }
94 if (c == '\n') {
95 fn(line); // also empty lines
96 line.clear();
97 }
98 }
99 if (!line.empty()) fn(line);
100 }
