14/**
15 * @brief Multi-line, word-wrapped text input box with scrolling.
16 *
17 * TextArea is TextField's multi-line counterpart - same focus/keyboard
18 * wiring, same TextInputTarget contract (see Core/TextInputTarget.h and
19 * TextField.h's class comment for the manage()/Keyboard pairing). Note that
20 * a Keyboard's dedicated Enter key (not Done) is what inserts a newline
21 * here - Done always finishes editing and closes the keyboard, exactly like
22 * TextField, via the same handleEnter()/submit() pair.
23 *
24 * The text is wrapped at word boundaries to the width of the box. When the
25 * text is taller than the box it scrolls: a drag over the box moves it, and
26 * while typing the end of the text is kept in view. The text is drawn clipped
27 * to the box.
28 */

109

116 }
117

120 const size_t radius = toPx(this->theme().shape.small);
121 target.fillRoundRect(toPx(box.x), toPx(box.y), toPx(box.w), toPx(box.h), radius,
122 this->theme().colors.surface);
123 target.drawRoundRect(toPx(box.x), toPx(box.y), toPx(box.w), toPx(box.h), radius, borderColor);

125 const int32_t pad = 8;
126 const bool showPlaceholder = text_.empty() && !focused_ && !placeholder_.empty();
127 const std::string& shown = showPlaceholder ? placeholder_ : text_;

129 tinygpu::IFont<RGB_T>& font = *this->theme().typography.body;
130 const int32_t lineH = static_cast<int32_t>(font.getHeight(1));

137 const int32_t contentH = static_cast<int32_t>(lines_.size()) * lineH;
138 maxScroll_ = contentH > viewH ? contentH - viewH : 0;
139 if (focused_ && !showPlaceholder && contentH - scrollY_ > viewH) scrollY_ = contentH - viewH;
140 clampScroll();
141
142 const RGB_T textColor = showPlaceholder ? (hasColors_ ? placeholderColor_ : this->theme().colors.onSurfaceVariant)
143 : (hasColors_ ? textColor_ : this->theme().colors.onSurface);
144 target.pushClipRect(toPx(box.x), toPx(box.y), toPx(box.w), toPx(box.h));
145 for (size_t i = 0; i < lines_.size(); i++) {
146 const int32_t y = box.y + pad + static_cast<int32_t>(i) * lineH - scrollY_;
147 font.drawText(target, static_cast<int16_t>(box.x + pad), static_cast<int16_t>(y), lines_[i].c_str(),
148 textColor, this->theme().colors.surface, false);
149 }
150 target.popClipRect();
151
152 // the caret after the last character
153 if (focused_ && cursorVisible_ && !showPlaceholder) {
154 const int32_t caretY = box.y + pad + static_cast<int32_t>(lines_.size() - 1) * lineH - scrollY_;

156 if (caretY >= box.y && caretY + lineH <= box.bottom()) {
157 target.drawLine(static_cast<size_t>(caretX + 1), static_cast<size_t>(caretY),
158 static_cast<size_t>(caretX + 1), static_cast<size_t>(caretY + lineH),
159 this->theme().colors.primary);
160 }

216 /// Breaks the text into lines no wider than maxW: at spaces, and at newlines.
217 /// A word wider than a line is split between characters.
218 static std::vector<std::string> wrap(const std::string& text, int32_t maxW,
219 tinygpu::IFont<RGB_T>& font) {
220 auto fits = [&](const std::string& s) {
221 return static_cast<int32_t>(font.measureTextWidth(s.c_str())) <= maxW;
222 };
223 std::vector<std::string> out;
224 std::string current;
225 std::string word;
226 for (size_t i = 0; i <= text.size(); i++) {
227 const char c = i < text.size() ? text[i] : '\n';
228 if (c != ' ' && c != '\n') {
229 word += c;
230 continue;
231 }
232 if (!word.empty()) {
233 const std::string sep = current.empty() ? "" : " ";
234 if (fits(current + sep + word)) {
235 current += sep + word;
236 } else {
237 if (!current.empty()) {
238 out.push_back(current);
239 current.clear();
240 }
241 current = word;
242 while (!fits(current) && current.size() > 1) {
243 size_t k = 1;
244 while (k < current.size() && fits(current.substr(0, k + 1))) k++;
245 out.push_back(current.substr(0, k));
246 current = current.substr(k);
247 }
248 }
249 word.clear();
250 }
251 if (c == '\n') {
252 out.push_back(current);
253 current.clear();
254 }
255 }
256 if (!current.empty()) out.push_back(current);
257 return out;
258 }
