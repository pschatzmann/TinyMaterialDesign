14/**
15 * @brief Multi-line, word-wrapped text input box.
16 *
17 * TextArea is TextField's multi-line counterpart - same focus/keyboard
18 * wiring, same TextInputTarget contract (see Core/TextInputTarget.h and
19 * TextField.h's class comment for the manage()/Keyboard pairing). Note that
20 * a Keyboard's dedicated Enter key (not Done) is what inserts a newline
21 * here - Done always finishes editing and closes the keyboard, exactly like
22 * TextField, via the same handleEnter()/submit() pair.
23 *
24 * Wrapping reuses TinyGPU's existing LinePrinter (the same one Card and
25 * Dialog use for their body text) rather than reimplementing word wrap. The
26 * blinking end-of-text cursor uses LinePrinter::cursorX()/cursorY(), which
27 * report where the next character would land after a print() call.
28 *
29 * Known limitation: no scrolling - text past the box's bottom edge is
30 * cropped to the box (see ISurface::pushClipRect()) rather than scrolling
31 * into view, and the cursor is skipped once it would fall past the box for
32 * the same reason. Fine for a few lines of notes; not a substitute for a
33 * real multi-page, scrollable text editor.
34 */

96 void draw(tinygpu::ISurface<RGB_T>& target) override {
97 const int32_t labelHeight = hasLabel_ ? 14 : 0;
98 const Bounds box(this->bounds.x, this->bounds.y + labelHeight, this->bounds.w,
99 this->bounds.h - labelHeight);
100
101 if (hasLabel_) {
102 RGB_T labelColor = focused_ ? this->theme().colors.primary : this->theme().colors.onSurfaceVariant;
103 if (!this->enabled) labelColor = blend(labelColor, this->theme().colors.surface, 0.5f);
104 this->theme().typography.label->drawText(target, static_cast<int16_t>(this->bounds.x),
105 static_cast<int16_t>(this->bounds.y), label_.c_str(),
106 labelColor, this->theme().colors.surface, false);
107 }
108
109 RGB_T borderColor = focused_ ? this->theme().colors.primary : this->theme().colors.outline;
110 if (!this->enabled) borderColor = blend(borderColor, this->theme().colors.surface, 0.5f);
111 const size_t radius = toPx(this->theme().shape.small);
112 target.fillRoundRect(toPx(box.x), toPx(box.y), toPx(box.w), toPx(box.h), radius,
113 this->theme().colors.surface);
114 target.drawRoundRect(toPx(box.x), toPx(box.y), toPx(box.w), toPx(box.h), radius, borderColor);
115
116 const int32_t pad = 8;
117 const bool showPlaceholder = text_.empty() && !focused_ && !placeholder_.empty();
118 const std::string& shown = showPlaceholder ? placeholder_ : text_;
119
120 // Defaults for an empty field - overwritten below once the printer
121 // actually runs, so the cursor still has somewhere sensible to sit.
122 size_t cursorX = toPx(box.x + pad);
123 size_t cursorY = toPx(box.y + pad);
124
125 if (!shown.empty()) {
126 const int32_t rightEdge = box.right() - pad;
127 const size_t rightBorder = rightEdge < static_cast<int32_t>(target.width())
128 ? target.width() - static_cast<size_t>(rightEdge)
129 : 0;
130
131 // Crops text past the box's bottom edge to the box (see the class
132 // comment) instead of letting it draw past it - LinePrinter/drawText
133 // have no notion of "this widget's own box", only the whole surface,
134 // so without this a long enough value would spill below the box.
135 target.pushClipRect(toPx(box.x), toPx(box.y), toPx(box.w), toPx(box.h));
136 tinygpu::LinePrinter<RGB_T> printer;
137 printer.setFont(*this->theme().typography.body);
138 printer.setTarget(target);
139 printer.setColor(showPlaceholder ? this->theme().colors.onSurfaceVariant : this->theme().colors.onSurface);
140 printer.setTopBorder(toPx(box.y + pad));
141 printer.setLeftBorder(toPx(box.x + pad));
142 printer.setRightBorder(rightBorder);
143 printer.print(shown.c_str());
144 cursorX = printer.cursorX();
145 cursorY = printer.cursorY();
146 target.popClipRect();
147 }
148
149 if (focused_ && cursorVisible_ && !showPlaceholder) {
150 const size_t lineHeight = this->theme().typography.body->getHeight(1);
151 if (cursorY + lineHeight <= toPx(box.bottom())) {
152 target.drawLine(cursorX + 1, cursorY, cursorX + 1, cursorY + lineHeight,
153 this->theme().colors.primary);
154 }
155 }
156 }

164 bool update(uint32_t nowMs) override {
165 const bool wasVisible = cursorVisible_;
166 if (!focused_) {
167 cursorVisible_ = false;

169 }
170 if (nowMs - lastBlinkMs_ >= kBlinkIntervalMs) {
171 cursorVisible_ = !cursorVisible_;

173 }

175 }
