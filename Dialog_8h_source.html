15/**
16 * @brief Modal alert dialog: full-screen scrim + centered card + title +
17 * wrapped message + action widgets (typically Button).
18 *
19 * Show it with Screen::presentDialog(dialog); while presented, Screen routes
20 * every gesture to the dialog first. Actions (usually Button) are not owned
21 * by Dialog - position each one's bounds (actionRect() suggests a spot),
22 * add it with addAction(), and wire its onClick to call
23 * screen.dismissDialog() itself.
24 *
25 * Alternative to addAction(): setActionProvider(count, at) switches to
26 * callback-driven actions instead - see Container.h's class comment for the
27 * same pattern (`at(index)` returns a Widget reference, typically from a
28 * small reused pool). Not usually needed for a dialog's handful of actions,
29 * but available for consistency.
30 */
31template <typename RGB_T = TINYMD_DEFAULT_RGB_T>

48 /// Registers `action` as content of this dialog. Ignored while an action
49 /// provider is active - see setActionProvider().
50 void addAction(Widget<RGB_T>& action) {
51 actions_.push_back(&action);
52 // Dialog might not have a theme yet if addAction() is called before
53 // the dialog itself is registered with Screen (via presentDialog()) -
54 // in that case setTheme() below will cascade to this action once it
55 // does.
56 if (this->theme_ != nullptr) action.setTheme(*this->theme_);
57 }

66 /// In provider mode, each action is themed at the point it's fetched

75 /// Suggested rect for the `index`-th of `count` actions, right-aligned
76 /// along the card's bottom edge.
77 Bounds actionRect(int index, int count, int32_t width = 80, int32_t height = 36) const {
78 const int32_t pad = 8;
79 const int32_t totalWidth = count * width + (count - 1) * pad;
80 const int32_t startX = this->bounds.right() - pad - totalWidth;
81 const int32_t y = this->bounds.bottom() - pad - height;
82 return Bounds(startX + index * (width + pad), y, width, height);
83 }

129 /// Screen routes every gesture here while the dialog is presented; only
130 /// taps landing on an action are forwarded to it, everything else is
131 /// swallowed (a modal dialog shouldn't leak touches through to whatever
132 /// is behind it).
133 bool onGesture(const tinygpu::GestureEvent& event) override {
134 const int count = actionCount();
135 for (int i = 0; i < count; ++i) {
136 Widget<RGB_T>* action = actionAt(i);
137 if (action->enabled && action->bounds.contains(event.point.x, event.point.y)) {
138 return action->onGesture(event);
139 }
140 }
141 return true;
142 }

144 bool update(uint32_t nowMs) override {
145 bool changed = false;
146 const int count = actionCount();
147 for (int i = 0; i < count; ++i) changed |= actionAt(i)->update(nowMs);
148 return changed;
149 }

151 /// True if there's an action at (x, y) - this dialog's own (screen-space)
152 /// coordinate space - that's draggable, recursing into it if it's itself
153 /// a composite - see Widget::isDraggableAt().
154 bool isDraggableAt(int32_t x, int32_t y) const override {
155 const int count = actionCount();
156 for (int i = 0; i < count; ++i) {
157 Widget<RGB_T>* action = actionAt(i);
158 if (action->enabled && action->bounds.contains(x, y)) return action->isDraggableAt(x, y);
159 }
160 return false;
161 }
162
163 private:
164 int actionCount() const {
165 return actionCountFn_ ? actionCountFn_() : static_cast<int>(actions_.size());
166 }
167
168 Widget<RGB_T>* actionAt(int index) const {
169 if (actionAtFn_) {
170 Widget<RGB_T>& action = actionAtFn_(index);
171 if (this->theme_ != nullptr) action.setTheme(*this->theme_);
172 return &action;
173 }
174 return actions_[static_cast<size_t>(index)];
175 }
