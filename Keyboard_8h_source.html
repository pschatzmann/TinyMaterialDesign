13/**
14 * @brief On-screen QWERTY keyboard that drives one text input at a time.
15 *
16 * A single Keyboard instance can serve any number of fields, TextField and
17 * TextArea interchangeably (see Core/TextInputTarget.h) - call manage(field)
18 * once per field to wire it up:
19 *
20 * Keyboard<RGB565> keyboard(Bounds(0, 420, 340, 190));
21 * keyboard.manage(nameField); // TextField
22 * keyboard.manage(notesField); // TextArea
23 *
24 * manage() makes tapping a field show this keyboard targeted at it (and
25 * blur whichever field it was previously targeting). Add the keyboard to
26 * Screen *after* every field it manages, so it draws (and hit-tests) on top
27 * of them while shown.
28 *
29 * Layout: a numbers row (which also holds the "123"/"ABC" key), three rows
30 * that switch between QWERTY letters and a page of common symbols
31 * (!@#$%^&*() / -_=+[]{}\\ / ;:'",./), and a control row (Shift / Space /
32 * Enter / Backspace / Done). Shift is a toggle (caps-lock style, not "one
33 * letter then auto-revert") and has no effect on the symbols page. Enter and
34 * Done are deliberately separate keys: Enter always inserts a literal '\n'
35 * and moves to the next line, regardless of field type; Done always means
36 * "finished editing" and closes the keyboard (see TextInputTarget::
37 * handleEnter(), which every field implements to return true) - so on a
38 * TextArea, Enter adds a line without leaving the keyboard, and Done closes
39 * it whenever you're actually done, mid-line or not. There is only this one
40 * symbols page (no further "more symbols" page) - a deliberate cut to keep
41 * this a two-layer keyboard rather than a deeper multi-page one.
42 */

52 /// Wires `field` so tapping it shows this keyboard targeted at it. `Field`
53 /// must implement TextInputTarget and expose a public
54 /// `std::function<void()> onFocus` - true of both TextField and TextArea.
55 template <typename Field>
56 void manage(Field& field) {
57 field.onFocus = [this, &field]() {
58 if (target_ != nullptr && target_ != &field) target_->blur();
59 target_ = &field;
60 this->visible = true;
61 };

70 forEachKey([&](const Key& key, const Bounds& rect) {

73 target.fillRoundRect(toPx(rect.x + 2), toPx(rect.y + 2), toPx(rect.w - 4), toPx(rect.h - 4),

88 this->theme().typography.body->drawText(target, static_cast<int16_t>(textX),

95 bool onGesture(const tinygpu::GestureEvent& event) override {
96 if (!isTapGesture(event.type)) return false;
97 const Key* hit = nullptr;
98 forEachKey([&](const Key& key, const Bounds& rect) {
99 if (rect.contains(event.point.x, event.point.y)) {
100 hit = &key;
101 return true; // stop iterating
102 }
103 return false;
104 });
105 if (hit == nullptr) return true; // swallow the tap either way
106 handleKey(*hit);
107 return true;
108 }

176 /// Walks every key, computing its Bounds from its row's weighted layout,
177 /// invoking `visitor(key, rect)`. Used by both draw() and onGesture() so
178 /// the layout math lives in exactly one place. `visitor` returns true to
179 /// stop iterating early (hit-testing); draw() always returns false.
180 /// Rows 1-3 swap between letters and symbols depending on mode_; row 0
181 /// (numbers + the mode toggle) and row 4 (controls) are the same either way.
182 template <typename Visitor>
183 void forEachKey(Visitor visitor) const {
184 struct Row {
185 const Key* keys;
186 size_t count;
187 };
188 const bool symbols = mode_ == Mode::kSymbols;
189 const Row rows[] = {
190 {kRow0, sizeof(kRow0) / sizeof(kRow0[0])},
191 symbols ? Row{kRow1Symbols, sizeof(kRow1Symbols) / sizeof(kRow1Symbols[0])}
192 : Row{kRow1, sizeof(kRow1) / sizeof(kRow1[0])},
193 symbols ? Row{kRow2Symbols, sizeof(kRow2Symbols) / sizeof(kRow2Symbols[0])}
194 : Row{kRow2, sizeof(kRow2) / sizeof(kRow2[0])},
195 symbols ? Row{kRow3Symbols, sizeof(kRow3Symbols) / sizeof(kRow3Symbols[0])}
196 : Row{kRow3, sizeof(kRow3) / sizeof(kRow3[0])},
197 {kRow4, sizeof(kRow4) / sizeof(kRow4[0])},
198 };
199 constexpr size_t kRowCount = sizeof(rows) / sizeof(rows[0]);
200 const int32_t rowHeight = this->bounds.h / static_cast<int32_t>(kRowCount);
201
202 for (size_t r = 0; r < kRowCount; ++r) {
203 float totalWeight = 0.0f;
204 for (size_t i = 0; i < rows[r].count; ++i) totalWeight += rows[r].keys[i].weight;
205
206 const int32_t rowY = this->bounds.y + static_cast<int32_t>(r) * rowHeight;
207 float cursor = 0.0f;
208 for (size_t i = 0; i < rows[r].count; ++i) {
209 const Key& key = rows[r].keys[i];
210 const int32_t x0 = this->bounds.x + static_cast<int32_t>(cursor / totalWeight * this->bounds.w);
211 cursor += key.weight;
212 const int32_t x1 = this->bounds.x + static_cast<int32_t>(cursor / totalWeight * this->bounds.w);
213 const Bounds rect(x0, rowY, x1 - x0, rowHeight);
214 if (visitor(key, rect)) return;
215 }
216 }
217 }

219 void handleKey(const Key& key) {
220 switch (key.action) {
221 case Action::kChar:
222 if (target_ != nullptr) {
223 target_->insertChar(shift_ ? static_cast<char>(std::toupper(key.ch)) : key.ch);
224 }
225 break;
226 case Action::kShift:
227 shift_ = !shift_;
228 break;
229 case Action::kBackspace:
230 if (target_ != nullptr) target_->backspace();
231 break;
232 case Action::kSpace:
233 if (target_ != nullptr) target_->insertChar(' ');
234 break;
235 case Action::kNewline:
236 // Unlike Done/handleEnter(), this always inserts a literal '\n' and
237 // moves to the next line, regardless of the target's field type.
238 if (target_ != nullptr) target_->insertChar('\n');
239 break;
240 case Action::kEnter:
241 // Both TextField and TextArea return true from handleEnter() -
242 // Done always finishes editing and closes the keyboard.
243 if (target_ == nullptr || target_->handleEnter()) this->visible = false;
244 break;
245 case Action::kToggleSymbols:
246 mode_ = (mode_ == Mode::kLetters) ? Mode::kSymbols : Mode::kLetters;
247 break;
248 }
249 }
