22/**
23 * @brief Open file dialog: browse folders and choose a file.
24 *
25 * The dialog does not read the file system itself. The application supplies
26 * listDirectory(), which returns the entries of a folder, so the same dialog
27 * works with SD cards, flash file systems and the desktop.
28 *
29 * The dialog is a panel (panel()) to show with Screen::presentDialog(). It
30 * lists the folders first, then the files whose extension is in the filter (all
31 * files if the filter is empty). A row "..", shown below the root, goes to the
32 * parent folder. Tapping a file calls onSelect(path); the application should
33 * dismiss the dialog then. Tapping Cancel calls onCancel (if set) and the
34 * dialog stays open unless the application dismisses it.
35 *
36 * @code
37 * FileDialog<RGB565> dialog;
38 * dialog.listDirectory = [](const std::string& dir) { ... SD entries ... };
39 * dialog.onSelect = [&](const std::string& path) { screen.dismissDialog(); open(path); };
40 * dialog.open(Bounds(8, 8, 224, 304), "Open book", "/", ".mobi,.azw,.azw3");
41 * screen.presentDialog(dialog.panel());
42 * @endcode
43 */

83 TypographyRole::kTitle);

85 TypographyRole::kLabel);
86 panel_.addChild(titleLabel_);
87 panel_.addChild(pathLabel_);
88
89 int32_t y = bounds_.y + 64;
90 const int32_t bottom = bounds_.bottom() - (onCancel ? 52 : 12);
91 if (dir_ != "/") {

93 y += kRowHeight;
94 }
95 std::vector<FileEntry> entries;
96 if (listDirectory) entries = listDirectory(dir_);
97 // folders first, then the files of the filter
98 for (int pass = 0; pass < 2; ++pass) {
99 for (const FileEntry& e : entries) {

102 if (!e.isDirectory && !matches(e.name)) continue;
103 const std::string name = e.name;

106 } else {

108 if (onSelect) onSelect(join(dir_, name));
109 });
110 }
111 y += kRowHeight;
112 }
113 }
114 if (rows_.empty()) {
115 addRow(y, "No files", nullptr);
116 rows_.back()->enabled = false;
117 }
118
119 if (onCancel) {
120 cancelButton_ = Button<RGB_T>(Bounds(bounds_.right() - 112, bottom + 8, 100, 36), "Cancel",
121 ButtonVariant::kText);
122 cancelButton_.onClick = [this]() {

138 bool matches(const std::string& name) const {
139 if (extensions_.empty()) return true;
140 size_t start = 0;
141 while (start <= extensions_.size()) {
142 size_t comma = extensions_.find(',', start);
143 if (comma == std::string::npos) comma = extensions_.size();
144 const std::string ext = extensions_.substr(start, comma - start);
145 if (name.size() >= ext.size() && name.compare(name.size() - ext.size(), ext.size(), ext) == 0) return true;
146 start = comma + 1;
147 }
148 return false;
149 }

151 static std::string join(const std::string& dir, const std::string& name) {
152 if (dir.empty() || dir.back() == '/') return dir + name;
153 return dir + "/" + name;

156 static std::string parentOf(const std::string& dir) {
157 const size_t slash = dir.find_last_of('/');
158 if (slash == std::string::npos || slash == 0) return "/";
159 return dir.substr(0, slash);
160 }
